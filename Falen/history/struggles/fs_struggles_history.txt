# Both regional struggles begin with the 2444 bookmark.
2444.1.1 = {
	effect = {
		if = {
			limit = { NOT = { exists = struggle:juche_struggle } }
			start_struggle = {
				struggle_type = juche_struggle
				start_phase = struggle_juche_phase_conquest
			}
		}
		if = {
			limit = { NOT = { exists = struggle:grandduke_struggle } }
			start_struggle = {
				struggle_type = grandduke_struggle
				start_phase = struggle_phase_grandduke_under_control
			}
		}
	}
}
