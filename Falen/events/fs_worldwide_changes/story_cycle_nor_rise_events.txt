# Save-load repair. New games normally create the struggle from history.
sun_from_north.9 = {
	type = character_event
	hidden = yes

	trigger = {
		NOT = { exists = struggle:juche_struggle }
		NOT = { has_global_variable = nk_rising }
		has_title = title:e_north_korea
	}

	immediate = {
		if = {
			limit = { current_date >= 2454.1.1 }
			start_struggle = {
				struggle_type = juche_struggle
				start_phase = struggle_juche_phase_status_quo
			}
		}
		else = {
			start_struggle = {
				struggle_type = juche_struggle
				start_phase = struggle_juche_phase_conquest
			}
		}
	}
}

# 2444: spread the prophecy and initialise its nine-year countdown.
sun_from_north.10 = {
	type = character_event
	hidden = yes

	trigger = {
		NOT = { has_global_variable = fs_juche_prophecy_started }
		has_title = title:e_north_korea
	}

	immediate = {
		# History normally creates the struggle. This also repairs starts made from
		# an older setup where history/struggles was missing.
		if = {
			limit = { NOT = { exists = struggle:juche_struggle } }
			start_struggle = {
				struggle_type = juche_struggle
				start_phase = struggle_juche_phase_conquest
			}
		}
		set_global_variable = { name = fs_juche_prophecy_started value = yes }
		set_global_variable = { name = fs_juche_prophecy_years value = 0 }
		capital_county = {
			add_county_modifier = {
				modifier = fs_juche_bunker_capital_garrison
				years = 20
			}
		}
		every_living_character = {
			limit = {
				location ?= { geographical_region = world_korea_north }
			}
			trigger_event = { id = sun_from_north.11 days = 1 }
		}
		# The first 10% roll is made on the 2444 start date. If it succeeds,
		# the old 2444.1.3 revival timing is retained as a rare outcome.
		random = {
			chance = 10
			trigger_event = { id = sun_from_north.1 days = 2 }
		}
		# Do not rely solely on the global yearly pulse: keep a self-scheduling
		# backup chain as well. Event 100 de-duplicates both paths.
		trigger_event = { id = sun_from_north.100 years = 1 }
	}
}

# Yearly controller. The 90 branch is also the annual phase incident.
sun_from_north.100 = {
	type = character_event
	hidden = yes

	trigger = {
		has_global_variable = fs_juche_prophecy_started
		NOT = { has_global_variable = nk_rising }
		NOT = { has_global_variable = fs_juche_annual_processed }
		has_title = title:e_north_korea
		current_date >= 2445.1.1
	}

	immediate = {
		# The global pulse and the self-scheduled chain can arrive on the same day.
		set_global_variable = {
			name = fs_juche_annual_processed
			value = yes
			days = 7
		}
		# Save-game repair: older versions set the prophecy variable without ever
		# constructing the struggle because its database/history paths were wrong.
		if = {
			limit = { NOT = { exists = struggle:juche_struggle } }
			start_struggle = {
				struggle_type = juche_struggle
				start_phase = struggle_juche_phase_conquest
			}
		}
		if = {
			# Twenty years after the 2444 start, the bunker must open.
			limit = { current_date >= 2464.1.1 }
			trigger_event = sun_from_north.1
		}
		else = {
			random_list = {
				10 = { trigger_event = sun_from_north.1 }
				90 = {
					if = {
						limit = { current_date >= 2454.1.1 }
						if = {
							limit = {
								struggle:juche_struggle = {
									is_struggle_phase = struggle_juche_phase_conquest
								}
							}
							struggle:juche_struggle = {
								change_struggle_phase = {
									struggle_phase = struggle_juche_phase_status_quo
								}
							}
							every_living_character = {
								limit = { location ?= { geographical_region = world_korea_north } }
								trigger_event = { id = sun_from_north.12 days = 1 }
							}
						}

						# Waiting: one adverse incident each year.
						random_list = {
							25 = {
								trigger = {
									any_county_in_region = {
										region = world_korea_north
										faith = faith:juche
										NOT = { this = title:c_jangjin }
									}
								}
								random_county_in_region = {
									region = world_korea_north
									limit = {
										faith = faith:juche
										NOT = { this = title:c_jangjin }
									}
									save_scope_as = juche_incident_county
									holder = { trigger_event = sun_from_north.23 }
								}
							}
							25 = {
								trigger = {
									any_ruler = {
										faith = faith:juche
										capital_province ?= { geographical_region = world_korea_north }
										NOT = { has_title = title:e_north_korea }
									}
								}
								random_ruler = {
									limit = {
										faith = faith:juche
										capital_province ?= { geographical_region = world_korea_north }
										NOT = { has_title = title:e_north_korea }
									}
									trigger_event = sun_from_north.24
								}
							}
							25 = {
								trigger = {
									any_county_in_region = {
										region = world_korea_north
										faith = faith:juche
									}
								}
								random_county_in_region = {
									region = world_korea_north
									limit = { faith = faith:juche }
									save_scope_as = juche_incident_county
									holder = { trigger_event = sun_from_north.25 }
								}
							}
							25 = {
								trigger = {
									any_ruler = {
										faith = faith:juche
										capital_province ?= { geographical_region = world_korea_north }
									}
								}
								random_ruler = {
									limit = {
										faith = faith:juche
										capital_province ?= { geographical_region = world_korea_north }
									}
									trigger_event = sun_from_north.26
								}
							}
						}
					}
					else = {
						# 2445-2453: nine yearly increments and one favourable incident.
						change_global_variable = { name = fs_juche_prophecy_years add = 1 }
						struggle:juche_struggle = {
							if = {
								limit = { is_struggle_phase = struggle_juche_phase_conquest }
								activate_struggle_catalyst = catalyst_juche_prophecy_year
							}
						}
						random_list = {
							34 = {
								trigger = {
									any_county_in_region = {
										region = world_korea_north
										NOT = { faith = faith:juche }
									}
								}
								random_county_in_region = {
									region = world_korea_north
									limit = { NOT = { faith = faith:juche } }
									save_scope_as = juche_incident_county
									holder = { trigger_event = sun_from_north.20 }
								}
							}
							33 = {
								trigger = {
									any_ruler = {
										faith = faith:juche
										capital_province ?= { geographical_region = world_korea_north }
									}
								}
								random_ruler = {
									limit = {
										faith = faith:juche
										capital_province ?= { geographical_region = world_korea_north }
									}
									trigger_event = sun_from_north.21
								}
							}
							33 = {
								trigger = {
									any_county_in_region = {
										region = world_korea_north
										faith = faith:juche
										holder = { NOT = { faith = faith:juche } }
									}
								}
								random_county_in_region = {
									region = world_korea_north
									limit = {
										faith = faith:juche
										holder = { NOT = { faith = faith:juche } }
									}
									save_scope_as = juche_incident_county
									holder = { trigger_event = sun_from_north.22 }
								}
							}
						}
					}
				}
			}
		}
		if = {
			limit = { NOT = { has_global_variable = nk_rising } }
			title:e_north_korea.holder ?= {
				trigger_event = { id = sun_from_north.100 years = 1 }
			}
		}
	}
}

sun_from_north.1 = {
	type = character_event
	title = sun_from_north.1.t
	desc = sun_from_north.1.desc
	theme = hunting
	override_background = {
		reference = sitting_room
	}
	left_portrait = root

	trigger = {
		NOT = { has_global_variable = nk_rising }
		has_title = title:e_north_korea
	}
	immediate = {
		set_global_variable = nk_rising
		if = {
			limit = { exists = struggle:juche_struggle }
			struggle:juche_struggle = {
				end_struggle = sun_from_north.1.t
			}
		}
		remove_global_variable = fs_juche_prophecy_years
		remove_global_variable = fs_juche_prophecy_started
		remove_global_variable = fs_juche_annual_processed
		root = { save_scope_as = avgvstvs }
	}
	option = {
		name = sun_from_north.1.a
		spawn_army = {
			uses_supply = no
			inheritable = yes
			name = Korean_People's_Army1
			levies = {
  				value = 0
				every_held_title = {
					limit = {
 						tier = tier_county
					}
					add = 1000
				}
			}
			location = root.capital_province
		}
		spawn_army = {
			uses_supply = no
			inheritable = yes
			name = Korean_People's_Army2
			levies = {
  				value = 0
				every_held_title = {
					limit = {
 						tier = tier_county
					}
					add = 1000
				}
			}
			location = root.capital_province
		}
		spawn_army = {
			uses_supply = no
			inheritable = yes
			name = Korean_People's_Army3
			levies = {
  				value = 0
				every_held_title = {
					limit = {
 						tier = tier_county
					}
					add = 1000
				}
			}
			location = root.capital_province
		}
		every_independent_ruler = {
			limit = {
				NOT = { this = root }
			}
			trigger_event = {
				id = sun_from_north.2
				days = 1
			}
		}
		every_player = {
			trigger_event = {
				id = sun_from_north.3
				days = 2
			}
		}
	}
}
