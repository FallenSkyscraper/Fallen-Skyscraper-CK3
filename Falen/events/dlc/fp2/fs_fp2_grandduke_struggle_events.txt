namespace = fs_grandduke_struggle

# Initialise the annual pulse when the struggle begins.
fs_grandduke_struggle.0001 = {
	type = character_event
	hidden = yes

	trigger = {
		exists = struggle:grandduke_struggle
		has_title = title:k_n_grandduke
		NOT = { has_global_variable = fs_grandduke_struggle_annual_chain }
	}

	immediate = {
		set_global_variable = { name = fs_grandduke_struggle_annual_chain value = yes }
		trigger_event = { id = fs_grandduke_struggle.0100 years = 1 }
	}
}

# One passive point per year and a chance for one Mechanist flavour event.
fs_grandduke_struggle.0100 = {
	type = character_event
	hidden = yes

	trigger = {
		exists = struggle:grandduke_struggle
		struggle:grandduke_struggle = {
			is_struggle_phase = struggle_phase_grandduke_under_control
		}
		has_title = title:k_n_grandduke
		current_date >= 2445.1.1
		NOT = { has_global_variable = fs_grandduke_year_processed }
	}

	immediate = {
		set_global_variable = {
			name = fs_grandduke_year_processed
			value = yes
			days = 7
		}

		struggle:grandduke_struggle = {
			activate_struggle_catalyst = {
				catalyst = catalyst_grandduke_year_passed
				character = root
			}
		}

		random = {
			chance = 40
			random_ruler = {
				limit = {
					is_adult = yes
					is_imprisoned = no
					faith = faith:mechanism
					capital_province ?= { geographical_region = custom_grandduke }
				}
				random_list = {
					33 = { trigger_event = fs_grandduke_struggle.2000 }
					33 = { trigger_event = fs_grandduke_struggle.2010 }
					34 = { trigger_event = fs_grandduke_struggle.2020 }
				}
			}
		}

		if = {
			limit = { exists = struggle:grandduke_struggle }
			title:k_n_grandduke.holder ?= {
				trigger_event = { id = fs_grandduke_struggle.0100 years = 1 }
			}
		}
	}
}

# The progress bar has reached 1000: the Grand Duchy collapses.
fs_grandduke_struggle.1000 = {
	type = character_event
	title = fs_grandduke_struggle.1000.t
	desc = fs_grandduke_struggle.1000.desc
	theme = realm
	override_background = { reference = battlefield }

	immediate = {
		set_global_variable = { name = fs_grandduke_struggle_fallen value = yes }

		every_ruler = {
			limit = {
				capital_province ?= { geographical_region = custom_grandduke }
				NOT = { faith = faith:mechanism }
			}
			trigger_event = { id = fs_grandduke_struggle.1001 days = 1 }
		}

		title:k_n_grandduke.holder ?= {
			destroy_title = title:k_n_grandduke
		}

		if = {
			limit = { exists = struggle:grandduke_struggle }
			struggle:grandduke_struggle = {
				end_struggle = fs_grandduke_struggle_fall_reason
			}
		}
		remove_global_variable = fs_grandduke_struggle_annual_chain
		remove_global_variable = fs_grandduke_year_processed
	}

	option = { name = fs_grandduke_struggle.1000.a }
}

# Every AI ruler takes the only AI-valid conversion option.
fs_grandduke_struggle.1001 = {
	type = character_event
	title = fs_grandduke_struggle.1001.t
	desc = fs_grandduke_struggle.1001.desc
	theme = faith

	option = {
		name = fs_grandduke_struggle.1001.a
		set_character_faith = faith:mechanism
		ai_chance = { base = 100 }
	}
	option = {
		name = fs_grandduke_struggle.1001.b
		trigger = { is_ai = no }
	}
}
