	title = fs_constituency_election.100.t
	desc = {
		desc = fs_constituency_election.100.intro

			first_valid = {
				triggered_desc = { trigger = { exists = scope:fs_constituency_result_winner } desc = fs_constituency_election.100.winner }
				desc = fs_constituency_election.100.empty
			}
		}
		desc = {
			first_valid = {
				triggered_desc = { trigger = { scope:fs_constituency_result_candidate ?= root scope:fs_constituency_candidate_reward > 0 } desc = fs_constituency_election.100.candidate }
				desc = fs_constituency_election.100.observer
			}
		}
		desc = { triggered_desc = { trigger = { scope:fs_constituency_host_reward > 0 } desc = fs_constituency_election.100.host_reward } }
		desc = { triggered_desc = { trigger = { scope:fs_constituency_candidate_reward > 0 } desc = fs_constituency_election.100.candidate_reward } }
		desc = { triggered_desc = { trigger = { scope:fs_constituency_winner_reward > 0 } desc = fs_constituency_election.100.winner_reward } }
		desc = { triggered_desc = { trigger = { scope:fs_constituency_observer_reward > 0 } desc = fs_constituency_election.100.observer_reward } }
		desc = { triggered_desc = { trigger = { scope:fs_constituency_result_fraud_count > 0 } desc = fs_constituency_election.100.scandal } }
		desc = { triggered_desc = { trigger = { scope:fs_constituency_result_covered_count > 0 } desc = fs_constituency_election.100.covered } }
	}
	theme = debate
	override_background = { reference = tgp_examination_room }
	left_portrait = { character = root animation = thinking }

		trigger = { exists = scope:fs_constituency_result_winner root != scope:fs_constituency_result_winner }
		character = scope:fs_constituency_result_winner
		animation = personality_bold
	}
	option = { name = fs_constituency_election.100.a }
}
