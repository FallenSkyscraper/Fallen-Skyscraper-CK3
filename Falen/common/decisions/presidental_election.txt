	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_invite_knights.dds"
	}
	ai_check_interval = 10


	desc = run_for_presidency_decision_desc
	selection_tooltip = run_for_presidency_decision_tooltip

	is_shown = {
		is_ai = yes
		is_imprisoned = no
		fs_prk_election_is_idle_trigger = yes
		OR = {
			AND = {
				is_ruler = yes
				is_landed = yes
				is_independent_ruler = no
				top_liege = {
					has_title = title:k_prk
				}
			}
			is_primary_heir_of = title:k_prk.holder
			has_title = title:k_prk
		}
		NOT = { has_variable = fs_prk_candidate }
	}
	is_valid = {
		fs_prk_can_register_election_candidate_trigger = yes
		fs_prk_ai_candidate_slot_available_trigger = yes
		prestige_level >= 1
	}
	cost = { gold = 60 prestige = 100 }
	effect = {
		fs_prk_register_election_candidate_effect = yes
	}
	ai_potential = {
		always = yes
	}
	ai_will_do = {
		base = 5

		modifier = {
			add = {
				value = ai_boldness
				multiply = 0.2
			}
		}
		modifier = {
			add = {
				value = ai_energy
				multiply = 0.5
			}
		}
		modifier = {
			add = 200
			has_trait = ambitious
		}
		modifier = {
			add = -100
			has_trait = content
		}
		modifier = {
			add = ai_sociability
		}
		modifier = {
			add = {
				value = ai_honor
				multiply = -0.2
			}
		}
		modifier = {
			add = 200
			has_title = title:k_prk
		}
	}
}
delay_election_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_invite_knights.dds"
	}
	ai_check_interval = 10

	desc = delay_election_decision_desc
	
	selection_tooltip = delay_election_decision_tooltip

	is_shown = {
		is_imprisoned = no
		has_title = title:k_prk
	}
	is_valid = {
		custom_description = {
			text = fs_prk_election_delay_requires_idle_tt
			fs_prk_election_is_idle_trigger = yes
			NOT = { has_global_variable = fs_prk_special_election_pending }
			NOT = { has_global_variable = fs_prk_special_election_active }
		}
		custom_description = {
			text = fs_prk_election_delay_once_per_cycle_tt
			NOT = { has_global_variable = fs_prk_election_delay_used }
		}
		global_var:election_term >= 0
		prestige_level >= 1
	}
	cost = { prestige = 400 }
	effect = {
		fs_prk_delay_election_calendar_effect = yes
		custom_tooltip = delay_election_decision_tt
		if = {
			limit = { is_at_war = yes }
			every_vassal = {
				add_opinion = {
					modifier = dictactor_opinion
					opinion = -10 # 전시면 좀 봐줌
					target = prev
				}
			}
		}
		else = {
			custom_tooltip = delay_election_decision_tt2
			hidden_effect = { add_prestige_level = -1 }
			every_vassal = {
				add_opinion = {
					modifier = dictactor_opinion
					target = prev
				}
			}
		}
	}
	ai_potential = {
		always = yes
	}
	ai_will_do = {
		base = 0
	}
}
