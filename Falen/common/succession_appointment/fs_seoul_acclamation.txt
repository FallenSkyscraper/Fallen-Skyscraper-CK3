seoul_acclamation = {
	default_candidates = { landed_vassal }
	allow_children = no
	allow_same_tier_candidates = no

	candidate_score = {
		value = {
			add = { value = diplomacy multiply = 2 desc = seoul_acclamation_diplomacy_score }
			add = { value = martial multiply = 2 desc = seoul_acclamation_martial_score }
			add = { value = stewardship multiply = 3 desc = seoul_acclamation_stewardship_score }
			add = { value = intrigue desc = seoul_acclamation_intrigue_score }
			add = { value = learning desc = seoul_acclamation_learning_score }
			add = {
				value = influence_level
				multiply = 20
				desc = seoul_acclamation_influence_score
			}
			if = {
				limit = {
					scope:title.holder = { any_councillor ?= { this = root } }
				}
				add = { value = 20 desc = seoul_acclamation_councillor_score }
			}
			if = {
				limit = { has_trait = incapable }
				add = { value = -1000 desc = seoul_acclamation_incapable_score }
			}
			if = {
				limit = {
					house ?= {
						has_house_aspiration_parameter = seoul_mayor_score_bonus
					}
				}
				add = { value = 20 desc = seoul_mayor_score_bonus }
			}
			if = {
				limit = {
					house ?= {
						has_house_aspiration_parameter = seoul_mayor_score_bonus_big
					}
				}
				add = { value = 40 desc = seoul_mayor_score_bonus_big }
			}
			if = {
				limit = {
					house ?= {
						has_house_aspiration_parameter = seoul_mayor_score_malus
					}
				}
				subtract = { value = 25 desc = seoul_mayor_score_malus }
			}
			if = {
				limit = {
					house ?= {
						has_house_aspiration_parameter = seoul_mayor_score_malus_2
					}
				}
				subtract = { value = 50 desc = seoul_mayor_score_malus_2 }
			}
			if = {
				limit = {
					house ?= {
						has_house_aspiration_parameter = seoul_mayor_score_malus_3
					}
				}
				subtract = { value = 100 desc = seoul_mayor_score_malus_3 }
			}
		}
	}
}
