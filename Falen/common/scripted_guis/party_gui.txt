party_gui_toggle = {
	scope = character
	
	is_shown = {
		has_variable = party_gui_toggle
	}

	effect = {
		if = {
			limit = {
				has_variable = party_gui_toggle
			}
			remove_variable = party_gui_toggle
		}
		else = {
			set_variable = party_gui_toggle
			if = {
				limit = { has_variable = hec_gui_toggle }
				remove_variable = hec_gui_toggle
			}
			if = {
				limit = { has_variable = prk_gui_toggle }
				remove_variable = prk_gui_toggle
			}
		}
	}
}

party_gui_close = {
	scope = character
	
	is_shown = {
		always = yes
	}

	effect = {
		remove_variable = party_gui_toggle
	}
}
party_gui_abc = {
	scope = character
	is_shown = {
		always = yes
	}
}
