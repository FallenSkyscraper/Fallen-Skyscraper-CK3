party_gui_toggle = {
	scope = character
	
	is_shown = {
		has_variable = party_gui_toggle
	}

	effect = {
		if = {
			limit = {
				has_variable = party_gui_toggle
			}
			remove_variable = party_gui_toggle
		}
		else = {
			set_variable = party_gui_toggle
			if = {
				limit = { has_variable = hec_gui_toggle }
				remove_variable = hec_gui_toggle
			}
			if = {
				limit = { has_variable = prk_gui_toggle }
				remove_variable = prk_gui_toggle
			}
		}
	}
}

party_gui_close = {
	scope = character
	
	is_shown = {
		always = yes
	}

	effect = {
		remove_variable = party_gui_toggle
	}
}
party_gui_abc = {
	scope = character
	is_shown = {

	}
}
is_partisan = {
	scope = character
	is_valid = {
		custom_tooltip = {
			text = should_have_party_affiliation
			has_variable = party_affiliation
		}
	}
}

party_creation_open_gui = {
	scope = character
	is_shown = { is_ai = no }
	is_valid = { NOT = { has_variable = party_creation_open } }
	effect = {
		if = {
			limit = {
				is_ai = no
				NOT = { has_variable = party_creation_open }
			}
			parties_clear_creation_draft_effect = yes
			set_variable = party_creation_open
			trigger_event = party_creation.0001
		}
	}
}

party_creation_select_communist_gui = {
	scope = character
	is_valid = {
		is_ai = no
		has_variable = party_creation_open
	}
	effect = {
		if = {
			limit = { is_ai = no has_variable = party_creation_open }
			set_variable = { name = party_creation_type value = flag:communist_party }
		}
	}
}
