						trigger_event = fs_grandduke_struggle.1000
					}
				}
				else = {
					random_ruler = {
						limit = { capital_province ?= { geographical_region = custom_grandduke } }
						trigger_event = fs_grandduke_struggle.1000
					}
				}
			}
		}
	}
