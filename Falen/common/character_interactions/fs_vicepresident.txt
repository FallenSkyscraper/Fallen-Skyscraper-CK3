# Liege (actor) selects one of their children to be their heir & diarch.
appoint_vice_president_interaction = {
	category = interaction_category_diarch
	icon = icon_declare_me_regent
	redirect = {
		scope:recipient = { save_scope_as = secondary_recipient }
		if = {
			limit = { scope:secondary_recipient ?= scope:actor }
			clear_saved_scope = secondary_recipient
		}
		if = {
			limit = {
				NOT = {
					scope:secondary_recipient ?= {
						OR = {
							is_courtier_of = scope:actor
							liege = scope:actor
						}
					}
				}
			}
			clear_saved_scope = secondary_recipient
		}
	}

	desc = appoint_vice_president_interaction_desc
	interface_priority = 280

	is_shown = {
		# Must be able to retain a co-emperor.
		scope:actor = { 
			government_has_flag = government_is_secular_republic
			highest_held_title_tier >= tier_kingdom
		}
		# We perform this upon ourselves or our (probable) candidates.
		OR = {
			scope:recipient ?= scope:actor
			scope:secondary_recipient ?= {
				OR = {
					is_courtier_of = scope:actor
					liege = scope:actor
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		# You can't already be in a diarchy of any kind.
		scope:actor = {
			has_active_diarchy = no
		}
		# Make sure we're targeting the right people, since redirects make this a lil slippery.
		trigger_if = {
			limit = { exists = scope:secondary_recipient }
			scope:secondary_recipient ?= {
				OR = {
					is_courtier_of = scope:actor
					liege = scope:actor
				}
			}
		}
		# Plus the usuals.
		scope:actor = { is_available_at_peace = yes }
	}

	populate_recipient_list = { sort_vice_president_candidates_to_list_effect = yes }

	cost = {
		prestige = {
			value = designate_heir_cost
			# Discount it because this _is_ worse than the standard path.
			multiply = 0.25
		}
	}

	auto_accept = yes

	pre_auto_accept = {
		# If we're an AI, explicitly score every valid candidate and randomize exact ties.
		scope:actor = {
			if = {
				limit = { is_ai = yes }
				sort_vice_president_candidates_to_list_effect = yes
				every_in_list = {
					list = characters
					fs_prk_calculate_ai_vice_president_score_effect = yes
				}
				ordered_in_list = {
					list = characters
					order_by = var:fs_prk_ai_vice_president_score
					position = 0
					save_scope_as = fs_prk_best_ai_vice_president
				}
				random_in_list = {
					list = characters
					limit = {
						var:fs_prk_ai_vice_president_score = scope:fs_prk_best_ai_vice_president.var:fs_prk_ai_vice_president_score
					}
					save_scope_as = ai_recipient
				}
				every_in_list = {
					list = characters
					remove_variable = fs_prk_ai_vice_president_score
				}
			}
		}
	}

	on_accept = {
		# Normalize the player and AI selections so every following effect uses one target.
		if = {
			limit = { exists = scope:ai_recipient }
			scope:ai_recipient = { save_scope_as = fs_prk_vice_president }
		}
		else_if = {
			limit = { exists = scope:secondary_recipient }
			scope:secondary_recipient = { save_scope_as = fs_prk_vice_president }
		}

		if = {
			limit = { exists = scope:fs_prk_vice_president }
			scope:actor = {
				# Always start the republic-specific diarchy, including for AI appointments.
				if = {
					limit = { is_ai = no }
					custom_tooltip = {
						text = appoint_co_monarch_interaction.tt.enter_co_monarchy_with_secondary_recipient
						try_start_diarchy = vice_presidency
						set_diarch = scope:fs_prk_vice_president
					}
				}
				else = {
					try_start_diarchy = vice_presidency
					set_diarch = scope:fs_prk_vice_president
				}

				# Preserve the old heir for opinion effects before overwriting them.
				designated_heir ?= {
					if = {
						limit = { this != scope:fs_prk_vice_president }
						if = {
							limit = { scope:actor = { is_ai = no } }
							custom_tooltip = appoint_co_monarch_interaction.tt.overwrite_designated_heir
						}
						save_scope_as = old_heir
					}
				}
				set_designated_heir = scope:fs_prk_vice_president
				# Keep an explicit pointer for the vacancy handler. The diarchy itself is
				# already dismantled by the time title inheritance on-actions finish.
				set_variable = {
					name = fs_prk_appointed_vice_president
					value = scope:fs_prk_vice_president
				}
			}

			# Apply appointment and displaced-officeholder opinions to the actual appointee.
			scope:fs_prk_vice_president = {
				set_variable = {
					name = co_emp_appointer
					value = scope:actor
				}
				add_opinion = {
					target = scope:actor
					modifier = made_co_ruler_opinion
					opinion = 50
				}
			}
			diarch_overwrites_old_designated_heir_apply_opinions_effect = {
				LIEGE = scope:actor
				CO_RULER = scope:fs_prk_vice_president
			}
			designated_diarch_overwridden_by_co_ruler_effect = {
				LIEGE = scope:actor
				CO_RULER = scope:fs_prk_vice_president
			}

			# Make sure the vice president is directly attached to the president.
			if = {
				limit = { scope:fs_prk_vice_president = { is_ruler = no } }
				scope:actor = { add_courtier = scope:fs_prk_vice_president }
			}
			if = {
				limit = { scope:fs_prk_vice_president.liege != scope:actor }
				scope:fs_prk_vice_president.liege = {
					add_opinion = {
						target = scope:actor
						modifier = retracted_vassal
					}
				}
				create_title_and_vassal_change = {
					type = revoked
					save_scope_as = change
				}
				scope:fs_prk_vice_president = {
					change_liege = {
						liege = scope:actor
						change = scope:change
					}
				}
				resolve_title_and_vassal_change = scope:change
			}

			# Object explorer logging.
			scope:actor = {
				diarch_interaction_set_or_update_global_variable_effect = { VARIABLE = di_track_appcomon_accept }
				diarch_interaction_add_to_global_list_effect = { VARIABLE_LIST = di_track_appcomon_accept_actor_list }
			}
			scope:fs_prk_vice_president = {
				diarch_interaction_add_to_global_list_effect = { VARIABLE_LIST = di_track_appcomon_accept_recipient_list }
			}
		}
	}

	# AI
	ai_targets = { ai_recipients = children }
	ai_targets = { ai_recipients = courtiers }
	ai_targets = { ai_recipients = vassals }
	ai_frequency = 12
	
	ai_potential = {
		government_has_flag = government_is_secular_republic
		highest_held_title_tier >= tier_kingdom
		has_active_diarchy = no
		# Do not require an adult child; any valid courtier or vassal may serve.
		OR = {
			any_courtier = { is_valid_basic_candidate_for_vizier_trigger = yes }
			any_vassal = { is_valid_basic_candidate_for_vizier_trigger = yes }
		}
	}
	
	ai_will_do = {
		base = 200
	}
}
