# Koreanism uses a fixed title picker instead of creating cardinalates.
fs_appoint_parliament_member_interaction = {
	category = interaction_category_religion
	icon = cardinal
	interface_priority = 45
	common_interaction = yes
	desc = fs_appoint_parliament_member_interaction_desc
	notification_text = fs_appoint_parliament_member_interaction_notification
	use_diplomatic_range = no
	ai_maybe = yes
	ai_min_reply_days = 1
	ai_max_reply_days = 5
	popup_on_receive = yes

	target_type = title
	target_source = actor
	target_filter = clerical_elector_titles
	show_all_targets = no

	is_available = {
		# has_pam_dlc_trigger = yes
		always = yes
	}
	is_shown = {
		scope:actor = {
			faith = faith:koreanism_faith
			has_title = title:k_koreanism
			this != scope:recipient
		}
		scope:recipient = {
			faith = faith:koreanism_faith
			NOT = { has_title = title:k_koreanism }
			NOT = { any_held_title = { fs_koreanism_parliament_seat_trigger = yes } }
		}
	}
	can_be_picked_title = {
		scope:target = { fs_koreanism_appointable_parliament_seat_trigger = yes }
	}
	is_valid_showing_failures_only = {
		scope:actor = { fs_koreanism_parliament_president_trigger = yes }
		custom_tooltip = {
			text = fs_parliament_candidate_tt
			scope:recipient = { fs_koreanism_parliament_candidate_trigger = yes }
		}
		custom_tooltip = {
			text = fs_parliament_vacancy_tt
			scope:actor = { fs_koreanism_has_parliament_vacancy_trigger = yes }
		}
	}
	has_valid_target_showing_failures_only = {
		custom_tooltip = {
			text = fs_parliament_selected_vacancy_tt
			exists = scope:target
			scope:target ?= { fs_koreanism_appointable_parliament_seat_trigger = yes }
		}
	}
	auto_accept = { scope:recipient = { is_ai = yes } }
	ai_accept = { base = 100 }
	on_accept = {
		# Recheck after the reply: another appointment may have filled the seat.
		if = {
			limit = {
				scope:actor = { fs_koreanism_parliament_president_trigger = yes }
				scope:recipient = { fs_koreanism_parliament_candidate_trigger = yes }
				exists = scope:target
				scope:target ?= { fs_koreanism_appointable_parliament_seat_trigger = yes }
			}
			scope:recipient = {
				fs_grant_koreanism_parliament_seat_effect = { SEAT = scope:target }
			}
		}
	}
	ai_potential = {
		fs_koreanism_parliament_president_trigger = yes
		fs_koreanism_has_parliament_vacancy_trigger = yes
	}
	ai_targets = {
		ai_recipients = courtiers
		ai_recipients = vassals
		ai_recipients = subordinates
		ai_recipients = clerical_region_rulers_in_realm
		max = 5
	}
	ai_target_quick_trigger = { adult = yes }
	ai_frequency = 6
	ai_will_do = {
		base = 20
		modifier = {
			add = 20
			scope:actor = { opinion = { target = scope:recipient value >= 50 } }
		}
		modifier = {
			factor = 0
			scope:actor = {
				OR = {
					has_relation_rival = scope:recipient
					has_relation_nemesis = scope:recipient
				}
			}
		}
	}
}

fs_dismiss_parliament_member_interaction = {
	category = interaction_category_religion
	icon = cardinal
	interface_priority = 45
	common_interaction = yes
	desc = fs_dismiss_parliament_member_interaction_desc
	use_diplomatic_range = no
	auto_accept = { always = yes }

	is_available = {
		# has_pam_dlc_trigger = yes
		always = yes
	}
	is_shown = {
		scope:actor = {
			faith = faith:koreanism_faith
			has_title = title:k_koreanism
			this != scope:recipient
		}
		scope:recipient = {
			NOT = { has_title = title:k_koreanism }
			any_held_title = { fs_koreanism_parliament_seat_trigger = yes }
		}
	}
	is_valid_showing_failures_only = {
		scope:actor = { fs_koreanism_parliament_president_trigger = yes }
		scope:recipient = {
			is_alive = yes
			NOT = { has_title = title:k_koreanism }
			any_held_title = { fs_koreanism_parliament_seat_trigger = yes }
		}
	}
	on_accept = {
		if = {
			limit = {
				scope:actor = { fs_koreanism_parliament_president_trigger = yes }
				scope:recipient = { NOT = { has_title = title:k_koreanism } }
			}
			scope:recipient = {
				every_held_title = {
					limit = { fs_koreanism_parliament_seat_trigger = yes }

						fs_vacate_koreanism_parliament_seat_effect = { SEAT = scope:fs_dismissed_parliament_seat }
					}
				}
				add_opinion = { target = scope:actor modifier = revoked_title }
			}
		}
	}
	ai_potential = { fs_koreanism_parliament_president_trigger = yes }
	ai_targets = {
		ai_recipients = papabile
		ai_recipients = subordinates
		ai_recipients = scripted_relations
	}
	ai_target_quick_trigger = { adult = yes }
	ai_frequency = 12
	ai_will_do = {
		base = 0
		modifier = {
			add = 30
			scope:actor = {
				OR = {
					has_relation_rival = scope:recipient
					has_relation_nemesis = scope:recipient
				}
			}
		}
		modifier = {
			add = 30
			scope:actor = { opinion = { target = scope:recipient value <= -50 } }
		}
		modifier = {
			factor = 0
			scope:actor = {
				OR = {
					has_relation_friend = scope:recipient
					has_relation_best_friend = scope:recipient
				}
			}
		}
	}
}


# Native interactions remain available to other faiths. Koreanism uses the fixed-seat actions above.
appoint_cardinal_interaction = {
	category = interaction_category_religion
	icon = cardinal
	interface_priority = 45
	common_interaction = yes

	# The Head of Faith names cardinals across all of Christendom
	use_diplomatic_range = no
	ai_maybe = yes
	ai_min_reply_days = 1
	ai_max_reply_days = 5
	popup_on_receive = yes
	notification_text = appoint_cardinal_interaction_notification

	desc = appoint_cardinal_interaction_desc

	is_available = {
		has_pam_dlc_trigger = yes
	}

	is_shown = {
		NOR = {
			scope:actor = { faith = faith:koreanism_faith }
			AND = {
				exists = scope:puppet_or_actor
				scope:puppet_or_actor = { faith = faith:koreanism_faith }
			}
		}
		scope:puppet_or_actor ?= {
			this != scope:recipient
			faith ?= { has_doctrine_parameter = has_clerical_electors }
			faith = scope:recipient.faith
			pam_leads_college_of_cardinals_trigger = yes
		}
		scope:recipient = {
			# is_clergy alone misses PAM's landed archbishops, whose organization does not match the faith's
			pam_is_investiture_clergy_trigger = yes
			pam_can_hold_cardinalate_trigger = yes
			pam_can_receive_cardinalate_trigger = yes
			pam_pope_is_head_of_faith_or_rite_trigger = yes
			NOT = { exists = clerical_elector_title } # already a cardinal
		}
	}

	cooldown_against_recipient = { years = 10 }

	is_valid_showing_failures_only = {
		trigger_if = {
			limit = { scope:is_puppet_action ?= yes }
			scope:puppet_or_actor = {
				piety >= medium_piety_value
			}
		}
		trigger_if = {
			limit = { scope:is_puppet_action ?= yes }
			scope:puppet_or_actor = {
				treasury >= minor_treasury_value
			}
		}
		custom_tooltip = {
			text = appoint_cardinal_interaction_max_cardinals_tt
			scope:puppet_or_actor.faith ?= {
				NOT = { any_clerical_elector_title = { count >= 70 } }
			}
		}
		scope:puppet_or_actor = {
			is_available_quick = {
				adult = yes
				incapable = no
				imprisoned = no
				hostage = no
			}
		}
		scope:recipient = {
			is_available_quick = {
				adult = yes
				incapable = no
				imprisoned = no
				hostage = no
			}
			NOR = {
				is_at_war_with = scope:puppet_or_actor
				has_trait = excommunicated
			}
		}
		custom_tooltip = {
			text = pam_interaction_clerical_gender_tt
			scope:puppet_or_actor.rite = { rite_has_allowed_gender_for_clergy = scope:recipient }
		}
	}

	cost = {
		treasury = {
			value = scope:actor.minor_treasury_value
			if = {
				limit = { scope:is_puppet_action ?= yes }
				multiply = 0
			}
		}
	}

	auto_accept = {
		trigger_if = {
			limit = {
				scope:recipient = {
					is_ai = yes
				}
			}
			always =  yes
		}
	}

	on_accept = {
		if = {
			limit = { scope:is_puppet_action ?= yes }
			scope:puppet_or_actor = {
				if = {
					limit = { minor_treasury_value > 0 }
					remove_treasury = {
						value = minor_treasury_value
					}
				}
			}
		}
		scope:recipient = {
			custom_tooltip = {
				text = appoint_cardinal_interaction_accept_tt
				hidden_effect = {
					grant_cardinal_title_effect = { CARDINAL_COUNTY = capital_county }
				}
			}
		}
		scope:puppet_or_actor = {
			show_as_tooltip = { add_piety = medium_piety_gain }
			hidden_effect = {
				send_interface_toast = {
					type = event_toast_effect_good
					title = appoint_cardinal_interaction_toast
					left_icon = scope:recipient
					right_icon = scope:cardinalate
					custom_tooltip = appoint_cardinal_interaction_accept_title_tt
					add_piety = medium_piety_value
				}
			}
			if = {
				limit = {
					scope:recipient = { pam_is_monastic_order_member_trigger = yes }
				}
				if = {
					limit = {
						can_add_hook = {
							target = scope:recipient
							type = favor_hook
						}
					}
					add_hook = {
						type = favor_hook
						target = scope:recipient
					}
				}
				pam_the_christian_church_appointed_cardinal_catalyst_effect = {
					POPE = scope:puppet_or_actor
					CARDINAL = scope:recipient
					OTHERWISE = catalyst_the_christian_church_appointed_monastic_cardinal
				}
			}
			else = {
				pam_the_christian_church_appointed_cardinal_catalyst_effect = {
					POPE = scope:puppet_or_actor
					CARDINAL = scope:recipient
					OTHERWISE = catalyst_the_christian_church_appointed_cardinal
				}
			}
		}
		if = {
			limit = {
				scope:is_puppet_action ?= yes
				scope:actor != scope:recipient
			}
			scope:actor = {
				send_interface_toast = {
					type = event_toast_effect_good
					title = appoint_cardinal_interaction_toast_puppeteer
					left_icon = scope:puppet_or_actor
					right_icon = scope:recipient
					custom_tooltip = appoint_cardinal_interaction_accept_title_tt
				}
			}
		}
	}

	on_decline = {
		scope:puppet_or_actor = {
			send_interface_toast = {
				type = event_toast_effect_bad
				title = appoint_cardinal_interaction_decline_toast
				right_icon = scope:recipient
				scope:recipient = {
					if = {
						limit = { is_ai = yes }
						add_opinion = {
							target = scope:puppet_or_actor
							modifier = annoyed_opinion
							opinion = -10
						}
					}
				}
			}
		}
		if = {
			limit = {
				scope:is_puppet_action ?= yes
				scope:actor != scope:recipient
			}
			scope:actor = {
				send_interface_toast = {
					type = event_toast_effect_bad
					title = appoint_cardinal_interaction_decline_toast_puppeteer
					left_icon = scope:puppet_or_actor
					right_icon = scope:recipient
				}
			}
		}
	}

	ai_accept = {
		base = 100
	}

	ai_targets = {
		ai_recipients = family
		max = 5
	}

	ai_targets = {
		ai_recipients = vassals
		ai_recipients = clerical_region_rulers_in_realm
		ai_recipients = rulers_in_clerical_region
		ai_recipients = subordinates
		ai_recipients = scripted_relations
		ai_recipients = faith_holy_order_leaders
	}

	ai_targets = {
		ai_recipients = courtiers
		max = 5
	}

	ai_target_quick_trigger = {
		adult = yes
	}

	ai_frequency_by_tier = {
		barony = 0
		county = 0
		duchy = 24
		kingdom = 6
		empire = 6
		hegemony = 6
	}

	ai_will_do = {
		base = 20
		modifier = { #The pope definitely wants to keep family close
			scope:puppet_or_actor = {
				is_any_family_relation_or_spouse_trigger = { CHARACTER = scope:recipient }
			}
			add = 70
		}
		modifier = {
			add = 10
			scope:recipient = {
				has_trait = overseer
			}
		}
		modifier = {
			add = 10
			scope:recipient = {
				has_trait = administrator
			}
		}
		modifier = {
			add = 30
			scope:recipient = {
				has_trait = zealous
			}
		}
		modifier = {
			add = 30
			scope:puppet_or_actor = {
				has_trait = zealous
			}
		}
		modifier = { # A cynical Pope still builds a College, just a venal one
			add = -20
			scope:puppet_or_actor = {
				has_trait = cynical
			}
		}
		modifier = {
			add = 20
			scope:puppet_or_actor = {
				is_allied_to = scope:recipient
			}
		}
		modifier = {
			add = 1000
			scope:puppet_or_actor = {
				has_relation_friend = scope:recipient

			scope:puppet_or_actor = {

			scope:recipient = {

					target = scope:puppet_or_actor

					holy_site_county ?= {
						holder ?= {

								this = scope:recipient
