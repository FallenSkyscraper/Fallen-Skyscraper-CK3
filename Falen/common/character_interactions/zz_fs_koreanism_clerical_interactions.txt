# Vanilla interactions with government-flag compatibility and faith-specific election titles.
request_ecclesiastical_title_creation_interaction = {
	category = interaction_category_religion
	interface_priority = 10
	desc = request_ecclesiastical_title_creation_interaction_desc
	notification_text = request_ecclesiastical_title_creation_interaction_notification
	icon = request_clerical_title_creation # TODO_PAM_ALI TIT-73245
	use_diplomatic_range = no

	# The picker chooses the seat, not the extent, and create_clerical_region fills the de jure kingdom
	target_type = title
	target_filter = domain_titles
	target_source = puppet_or_actor
	show_all_targets = no

	cost = {
		piety = {
			# Free Christian de jure county count of the anchor's de jure kingdom * piety
			add = {
				value = pam_new_clerical_region_cost_value
				if = { # AI pays less
					limit = { scope:actor = { is_ai = yes } }
					multiply = 0.75
				}
			}
			if = {
				limit = { scope:is_puppet_action ?= yes }
				multiply = 0
			}
		}
	}

	# Sets the new-holder slot (secondary_recipient): secular → realm priest (picker can override); clergy → themselves. Title is resolved in on_accept (redirect can't save titles)
	redirect = {
		if = {
			limit = {
				scope:puppet_or_actor ?= { government_has_flag = government_is_ecclesiastical }
			}
			scope:puppet_or_actor = {
				save_scope_as = secondary_recipient
			}
		}
		else = {
			scope:puppet_or_actor = {
				# A character may hold only one clerical region, so an existing archbishop cannot default here
				if = {
					limit = {
						exists = cp:councillor_court_chaplain
						cp:councillor_court_chaplain = {
							faith = scope:puppet_or_actor.faith
							NOT = { any_held_title = { has_clerical_region = yes } }
						}
					}
					cp:councillor_court_chaplain = {
						save_scope_as = secondary_recipient
					}
				}
				else = {
					ordered_courtier = {
						limit = {
							NOT = { any_held_title = { has_clerical_region = yes } }
							can_be_court_chaplain_trigger = { COURT_OWNER = scope:puppet_or_actor }
						}
						order_by = learning
						save_scope_as = secondary_recipient
					}
				}
			}
		}
	}

	# Holder picker (secular actors only) — realm priest is the pre-selected default via redirect
	populate_recipient_list = {
		scope:puppet_or_actor = {
			cp:councillor_court_chaplain ?= {
				if = {
					limit = {
						faith = scope:puppet_or_actor.faith
						NOT = { any_held_title = { has_clerical_region = yes } }
					}
					add_to_list = characters
				}
			}
			every_courtier = {
				limit = {
					NOT = { any_held_title = { has_clerical_region = yes } }
					can_be_court_chaplain_trigger = { COURT_OWNER = scope:puppet_or_actor }
				}
				add_to_list = characters
			}
			every_vassal = {
				limit = {
					NOT = { any_held_title = { has_clerical_region = yes } }
					can_be_court_chaplain_trigger = { COURT_OWNER = scope:puppet_or_actor }
					government_has_flag = government_is_ecclesiastical
				}
				add_to_list = characters
			}
		}
	}

	# can_be_picked_title handles title eligibility, so we keep is_shown fast for the AI
	is_shown = {
		scope:puppet_or_actor.faith ?= {
			has_doctrine_parameter = has_clerical_regions
			OR = {
				NOT = { has_doctrine_parameter = temporal_head_of_faith }
				NOT = { exists = religious_head }
			}
		}
		OR = {
			scope:puppet_or_actor ?= {
				highest_held_title_tier >= tier_duchy
				is_available_quick = {
					adult = yes
					alive = yes
					imprisoned = no
					incapable = no
				}
				NOT = { pam_government_is_religious_playable_trigger = yes }
				any_held_title = {
					tier >= tier_duchy
					is_landless_type_title = no
					is_head_of_faith = no
					title_capital_county ?= {
						NOT = { exists = clerical_region_title }
					}
				}
			}
			scope:puppet_or_actor ?= {
				government_has_flag = government_is_ecclesiastical
				is_landed = yes
				liege.primary_title ?= {
					tier = tier_kingdom
					is_landless_type_title = no
				}
				is_available_quick = {
					adult = yes
					alive = yes
					imprisoned = no
					incapable = no
				}
				primary_title ?= { tier < tier_duchy }
				NOT = { any_held_title = { has_clerical_region = yes } }
				any_held_title = {
					tier >= tier_county
					is_landless_type_title = no
					is_head_of_faith = no
					title_capital_county ?= {
						NOT = { exists = clerical_region_title }
					}
				}
			}
		}
		# has_pam_dlc_trigger = yes
		scope:recipient != scope:puppet_or_actor
		OR = {
			pam_recipient_is_clerical_title_authority_trigger = yes # HoF, or the Head of Rites when the faith has no HoF
			# A realm under an antipope asks them instead
			scope:puppet_or_actor.religious_head_or_challenger ?= scope:recipient
		}
	}

	is_valid_showing_failures_only = {
		trigger_if = {
			limit = { scope:is_puppet_action ?= yes }
			scope:puppet_or_actor = {
				piety >= pam_new_clerical_region_cost_value
			}
		}
		scope:puppet_or_actor = {
			is_available_quick = {
				imprisoned = no
				incapable = no
			}
			pam_antipope_clerical_request_trigger = { RECIPIENT = scope:recipient }
			NOT = {
				has_character_modifier = promised_pilgrimage_to_clergy_modifier
			}
		}
		custom_tooltip = {
			text = pam_request_ecclesiastical_title_creation_needs_free_cleric_tt
			exists = scope:secondary_recipient
			scope:secondary_recipient = {
				faith = scope:puppet_or_actor.faith
				NOT = { any_held_title = { has_clerical_region = yes } }
			}
		}
		trigger_if = {
			limit = { scope:actor = scope:puppet_or_actor }
			custom_tooltip = {
				text = pam_request_ecclesiastical_title_creation_no_eligible_anchor_tt
				scope:puppet_or_actor = {
					any_held_title = {
						pam_valid_new_clerical_region_anchor_trigger = { PETITIONER = scope:puppet_or_actor }
					}
				}
			}
		}
		trigger_else = {
			custom_tooltip = {
				text = pam_request_ecclesiastical_title_creation_no_eligible_anchor_tt_puppeteer
				scope:puppet_or_actor = {
					any_held_title = {
						pam_valid_new_clerical_region_anchor_trigger = { PETITIONER = scope:puppet_or_actor }
					}
				}
			}
		}
	}

	has_valid_target_showing_failures_only = {
		custom_tooltip = {
			text = pam_new_clerical_region_invalid_anchor_tt
			scope:target ?= {
				pam_valid_new_clerical_region_anchor_trigger = { PETITIONER = scope:puppet_or_actor }
			}
		}
	}

	can_be_picked_title = {
		scope:target = {
			pam_valid_new_clerical_region_anchor_trigger = { PETITIONER = scope:puppet_or_actor }
		}
	}

	is_highlighted = {
		scope:puppet_or_actor.piety_level > scope:recipient.piety_level
	}

	# Creates a clerical region on the capital of the picked title; appoints the new holder as court chaplain of that title's holder
	on_accept = {
		scope:secondary_recipient = {
			save_scope_as = new_ecclesiastical_holder
		}
		# The picked title seats the region; create_clerical_region fills it from that seat's de jure kingdom
		scope:target = {
			save_scope_as = target_kingdom
			holder = { save_scope_as = kingdom_holder }
		}
		# We bring the new holder into court as chaplain unless they already hold the seat
		if = {
			limit = { scope:kingdom_holder != scope:new_ecclesiastical_holder }
			scope:kingdom_holder = {
				if = {
					limit = {
						scope:new_ecclesiastical_holder = {
							NOT = { is_courtier_of = scope:kingdom_holder }
							is_ruler = no
						}
					}
					add_courtier = scope:new_ecclesiastical_holder
				}
				if = {
					limit = {
						scope:new_ecclesiastical_holder = {
							NOT = { is_councillor_of = scope:kingdom_holder }
						}
					}
					assign_councillor_type = {
						type = councillor_court_chaplain
						remove_existing_councillor = yes
						target = scope:new_ecclesiastical_holder
					}
				}
			}
		}
		hidden_effect = {
			if = {
				limit = {
					# has_pam_dlc_trigger = yes
					scope:puppet_or_actor = {
						religion = religion:christianity_religion
						any_character_situation = {
							situation_type = the_christian_church
							situation_top_has_catalyst = catalyst_the_christian_church_clerical_region_created
						}
					}
				}
				situation:the_christian_church = {
					trigger_situation_catalyst = catalyst_the_christian_church_clerical_region_created
				}
			}
		}
		# Create the clerical region on the kingdom's capital and give the new title to the new holder
		custom_tooltip = {
			text = request_create_new_ecclesiastical_title_effect_tt
			create_title_and_vassal_change = {
				type = created
				save_scope_as = change
				add_claim_on_loss = no
			}
			scope:target_kingdom = {
				title_capital_county = {
					save_scope_as = clerical_region_anchor_county
					random_county_province = {
						limit = { is_county_capital = yes }
						save_scope_as = region_capital
						if = {
							limit = { scope:new_ecclesiastical_holder = { faith = faith:koreanism_faith } }
							create_clerical_region = {
								domicile_type = admin_office
								name = CLERICAL_REGION_PROVINCE_TITLE_NAME
								save_scope_as = new_title
							}
						}
						else = {
							create_clerical_region = {
								domicile_type = ecclesiastical_domicile
								name = CLERICAL_REGION_PROVINCE_TITLE_NAME
								save_scope_as = new_title
							}
						}
						scope:new_title ?= { # ?= avoids a tooltip-time error since the title does not exist yet when the description is built
							set_coa = scope:target_kingdom
							set_color_from_title = scope:target_kingdom
							set_capital_county = scope:clerical_region_anchor_county
							set_landless_title = yes
							set_no_automatic_claims = yes
							set_always_follows_primary_heir = yes
							change_title_holder = {
								holder = scope:new_ecclesiastical_holder
								change = scope:change
							}
							resolve_title_and_vassal_change = scope:change
						}
					}
				}
			}
		}
		scope:actor = {
			send_interface_toast = {
				type = event_toast_effect_good
				title = request_ecclesiastical_title_creation_accepted_tt
				left_icon = scope:new_title
				right_icon = scope:new_ecclesiastical_holder
				# New holder is grateful — skipped when the petitioner appointed themselves (clergy path), to avoid a self-opinion bonus
				if = {
					limit = {
						scope:new_ecclesiastical_holder != scope:puppet_or_actor
					}
					reverse_add_opinion = {
						target = scope:new_ecclesiastical_holder
						modifier = grateful_opinion
						opinion = 50
					}
				}
			}
			if = {
				limit = {
					NOT = {
						government_has_flag = government_is_ecclesiastical
					}
				}
				add_hook = {
					type = loyalty_hook
					target = scope:new_ecclesiastical_holder
				}
			}
		}
		if = {
			limit = { scope:is_puppet_action ?= yes }
			scope:puppet_or_actor = {
				send_interface_toast = {
					type = event_toast_effect_good
					title = request_ecclesiastical_title_creation_accepted_tt
					left_icon = scope:new_title
					right_icon = scope:new_ecclesiastical_holder
				}
			}
		}
		if = {
			limit = { scope:recipient != scope:actor }
			scope:recipient = {
				send_interface_toast = {
					type = event_toast_effect_good
					title = request_ecclesiastical_title_creation_accepted_tt
					left_icon = scope:new_title
					right_icon = scope:new_ecclesiastical_holder
				}
			}
		}
		# When a puppeteer initiates, the puppet also drains their own piety up to their current maximum
		scope:puppet_or_actor = {
			if = {
				limit = {
					piety > 0
					scope:is_puppet_action ?= yes
					pam_new_clerical_region_cost_value > 0
				}
				add_piety = {
					value = {
						add = pam_new_clerical_region_cost_value
						max = piety
						multiply = -1
					}
				}
			}
		}
		scope:new_ecclesiastical_holder = {
			if = {
				limit = {
					is_ruler = yes # Holder becomes a ruler only after resolve_title_and_vassal_change; guard prevents a tooltip crash
					NOT = { government_has_flag = government_is_ecclesiastical }
				}
				fs_set_clerical_office_government_effect = yes
			}
			faith = {
				change_fervor = {
					value = notable_fervor_gain
					desc = fervor_gain_clerical_title_created
				}
			}
		}
	}

	on_decline = {
		scope:puppet_or_actor = {
			if = {
				limit = { is_ai = yes }
				add_opinion = {
					target = scope:recipient
					modifier = annoyed_opinion
					opinion = -10
				}
			}
		}
	}

	ai_targets = {
		ai_recipients = head_of_faith
		ai_recipients = head_of_rite
	}

	ai_frequency_by_tier = {
		barony = 0
		county = 36
		duchy = 36
		kingdom = 36
		empire = 36
		hegemony = 36
	}

	ai_accept = {
		base = 75

		celestial_hierarchy_acceptance_modifier = {
			ACTOR = scope:puppet_or_actor
			RECIPIENT = scope:recipient
		}

		clerical_petition_to_authority_acceptance_modifier = yes

		# HOF's opinion of the new holder
		opinion_modifier = {
			trigger = {
				exists = scope:secondary_recipient
				scope:secondary_recipient != scope:puppet_or_actor
			}
			who = scope:recipient
			opinion_target = scope:secondary_recipient
			desc = AI_OPINION_REASON
		}

		# New holder's rite differs from the HoF
		modifier = {
			add = -25
			exists = scope:secondary_recipient
			scope:secondary_recipient != scope:puppet_or_actor
			scope:secondary_recipient.rite != scope:recipient.rite
			desc = SECONDARY_RECIPIENT_DIFFERENT_RITE
		}

		# New holder has virtuous traits
		modifier = {
			add = {
				add = 10
				multiply = scope:secondary_recipient.num_virtuous_traits
			}
			exists = scope:secondary_recipient
			scope:secondary_recipient != scope:puppet_or_actor
			scope:secondary_recipient.num_virtuous_traits > 0
			desc = SECONDARY_RECIPIENT_VIRTUOUS
		}

		# New holder has sinful traits
		modifier = {
			add = {
				add = -10
				multiply = scope:secondary_recipient.num_sinful_traits
			}
			exists = scope:secondary_recipient
			scope:secondary_recipient != scope:puppet_or_actor
			scope:secondary_recipient.num_sinful_traits > 0
			desc = SECONDARY_RECIPIENT_SINFUL
		}

		# New holder has a learning education
		modifier = {
			add = 10
			exists = scope:secondary_recipient
			scope:secondary_recipient != scope:puppet_or_actor
			scope:secondary_recipient = { has_education_learning_trigger = yes }
			desc = SECONDARY_RECIPIENT_LEARNING_EDUCATION_REASON
		}

		# New holder is celibate
		modifier = {
			add = 15
			exists = scope:secondary_recipient
			scope:secondary_recipient != scope:puppet_or_actor
			scope:secondary_recipient = { has_trait = celibate }
			desc = SECONDARY_RECIPIENT_CELIBATE
		}

		# New holder is devoted
		modifier = {
			add = 15
			exists = scope:secondary_recipient
			scope:secondary_recipient != scope:puppet_or_actor
			scope:secondary_recipient = { has_trait = devoted }
			desc = SECONDARY_RECIPIENT_DEVOTED
		}

		# New holder shares the realm's culture
		modifier = {
			add = 10
			exists = scope:secondary_recipient
			scope:secondary_recipient != scope:puppet_or_actor
			scope:secondary_recipient.culture = scope:puppet_or_actor.capital_county.culture
			desc = SECONDARY_RECIPIENT_SAME_CULTURE_AS_SEAT
		}
	}

	ai_will_do = {
		base = 100 # These have high costs, kind of steep requirements and we always want the AI to expand or create clerical titles whenever available, if they can convince the HoF

		# Ambitious — wants to establish or expand clerical presence in their realm
		modifier = {
			trigger = {
				scope:puppet_or_actor = { has_trait = ambitious }
			}
			add = 50
		}

		# Zealous — eager to plant the faith on new soil
		modifier = {
			trigger = {
				scope:puppet_or_actor = { has_trait = zealous }
			}
			add = 50
		}

		# Secular: incentive increases when the realm has ANY uncovered de jure counties
		modifier = {
			trigger = {
				scope:puppet_or_actor = {
					NOT = { government_has_flag = government_is_ecclesiastical }
					any_held_title = {
						tier >= tier_duchy
						any_de_jure_county = {
							NOT = { exists = clerical_region_title }
						}
					}
				}
			}
			add = 50
		}

		# The seat's de jure kingdom already has a see, so this one can only cover whatever pockets it could not reach
		modifier = {
			trigger = {
				exists = scope:target.title_capital_county.kingdom
				scope:target.title_capital_county.kingdom = {
					any_de_jure_county = {
						exists = clerical_region_title
					}
				}
			}
			add = -75
		}

		# Clergy: dislikes their secular liege — wants an independent title of their own
		opinion_modifier = {
			trigger = {
				scope:puppet_or_actor = {
					government_has_flag = government_is_ecclesiastical
					exists = liege
				}
			}
			opinion_target = scope:puppet_or_actor.liege
			who = scope:puppet_or_actor
			multiplier = -0.5
		}

		# Clergy: liege dislikes them back — reinforces their motivation to break away
		opinion_modifier = {
			trigger = {
				scope:puppet_or_actor = {
					government_has_flag = government_is_ecclesiastical
					exists = liege
				}
			}
			opinion_target = scope:puppet_or_actor
			who = scope:puppet_or_actor.liege
			multiplier = -0.5
		}
	}
}

request_ecclesiastical_title_split_interaction = {
	category = interaction_category_religion
	interface_priority = 10
	desc = request_ecclesiastical_title_split_interaction_desc
	notification_text = request_ecclesiastical_title_split_interaction_notification
	icon = request_clerical_title_split # TODO_PAM_ALI TIT-73560
	use_diplomatic_range = no

	cooldown = { years = 10 }

	cost = {
		piety = {
			# Province count of the liege's primary title * piety
			add = {
				value = pam_request_ecclesiastical_title_split_cost_value
				if = { # AI pays half
					limit = { scope:actor = { is_ai = yes } }
					multiply = 0.75
				}
			}
			if = {
				limit = { scope:is_puppet_action ?= yes }
				multiply = 0
			}
		}
	}

	# secondary_actor = new holder (secular: realm priest, picker can override; clergy: themselves); secondary_recipient = affected superior/region holder. Both slots persist into ai_accept/on_accept
	redirect = {
		# New holder of the split-off region
		if = {
			limit = {
				scope:puppet_or_actor ?= { is_clergy = yes }
			}
			scope:puppet_or_actor = {
				save_scope_as = secondary_actor
			}
		}
		else = {
			scope:puppet_or_actor = {
				cp:councillor_court_chaplain ?= {
					save_scope_as = secondary_actor
				}
			}
		}
		# Affected clerical superior / region holder — stored in the secondary_recipient slot so it persists into ai_accept and on_accept
		if = {
			limit = {
				scope:puppet_or_actor ?= {
					NOT = { pam_government_is_religious_playable_trigger = yes }
					is_landed = yes
					highest_held_title_tier >= tier_duchy
					capital_county = { exists = clerical_region_title }
				}
			}
			scope:puppet_or_actor.capital_county.clerical_region_title.holder ?= {
				save_scope_as = secondary_recipient
			}
		}
		else = {
			scope:puppet_or_actor.superior ?= {
				save_scope_as = secondary_recipient
			}
		}
	}

	# New-holder picker — realm priest is the pre-selected default for secular rulers; clergy default to themselves
	populate_actor_list = {
		scope:puppet_or_actor = {
			# Clergy can take the new title themselves (the redirect default)
			if = {
				limit = { is_clergy = yes }
				add_to_list = characters
			}
			cp:councillor_court_chaplain ?= { add_to_list = characters }
			every_courtier = {
				limit = {
					NOT = { any_held_title = { has_clerical_region = yes } }
					can_be_court_chaplain_trigger = { COURT_OWNER = scope:puppet_or_actor }
				}
				add_to_list = characters
			}
			every_vassal = {
				limit = {
					NOT = { any_held_title = { has_clerical_region = yes } }
					can_be_court_chaplain_trigger = { COURT_OWNER = scope:puppet_or_actor }
					government_has_flag = government_is_ecclesiastical
				}
				add_to_list = characters
			}
		}
	}

	# Visible for secular duke+ rulers whose capital sits in a splittable clerical region, and for landed county-tier clergy splitting off from their superior
	is_shown = {
		scope:puppet_or_actor.faith ?= {
			has_doctrine_parameter = has_clerical_regions
			OR = {
				NOT = { has_doctrine_parameter = temporal_head_of_faith }
				NOT = { exists = religious_head }
			}
		}
		OR = {
			# Clergy branch: county-tier (or landless) clergy splitting off from their superior
			scope:puppet_or_actor ?= {
				is_clergy = yes
				trigger_if = {
					limit = {
						government_has_flag = government_is_ecclesiastical
						is_landed = yes
					}
					highest_held_title_tier = tier_county
				}
				exists = superior
			}
			# Secular branch: duke+ secular ruler whose capital is in a held clerical region
			scope:puppet_or_actor ?= {
				highest_held_title_tier >= tier_duchy
				is_landed = yes
				capital_county ?= {
					clerical_region_title ?= { exists = holder }
				}
				NOT = { pam_government_is_religious_playable_trigger = yes }
			}
		}
		scope:puppet_or_actor ?= {
			is_available_quick = {
				adult = yes
				alive = yes
				imprisoned = no
				incapable = no
			}
		}
		# has_pam_dlc_trigger = yes
		scope:recipient != scope:puppet_or_actor
		OR = {
			pam_recipient_is_clerical_title_authority_trigger = yes # HoF, or the Head of Rites when the faith has no HoF
			# A realm under an antipope asks them instead
			scope:puppet_or_actor.religious_head_or_challenger ?= scope:recipient
		}
		scope:recipient = {
			trigger_if = {
				limit = { scope:actor = { is_ai = yes } }
				NOT = { has_variable = pam_request_ecclesiastical_title_split_ai_cooldown }
			}
		}
	}

	is_valid_showing_failures_only = {

				text = pam_split_too_small_tt
				scope:puppet_or_actor ?= { split_clerical_region_size >= 3 }

			# The new region must not be larger than what the superior keeps (at most half of the region)
			custom_tooltip = {
				text = pam_split_larger_than_superior_tt
				scope:puppet_or_actor ?= {
					split_clerical_region_size <= {
						value = capital_county.clerical_region_title.clerical_region_size
						divide = 2
					}
				}
			}
		}
		trigger_else = {
			custom_tooltip = {
				text = pam_split_region_is_religious_head_seat_tt
				scope:puppet_or_actor.superior ?= {
					this != scope:puppet_or_actor.religious_head_or_challenger
					this != rite.head_of_rite
				}
			}
			# Clergy: must sit under a different secular liege than their superior does
			trigger_if = {
				limit = { scope:actor = scope:puppet_or_actor }
				custom_tooltip = {
					text = same_liege_as_superior_tt
					scope:puppet_or_actor.liege != scope:puppet_or_actor.superior.liege
				}
			}
			trigger_else = {
				custom_tooltip = {
					text = same_liege_as_superior_tt_puppeteer
					scope:puppet_or_actor.liege != scope:puppet_or_actor.superior.liege
				}
			}
			# The region's own capital stays with the original see; the clergy split founds the new title on the liege's capital, so the same seat collision orphans counties
			custom_tooltip = {
				text = pam_split_liege_capital_is_region_seat_tt
				scope:puppet_or_actor.liege.capital_county ?= {
					NOT = { this = clerical_region_title.title_preferred_capital_county }
				}

			# No tiny splits — the new region needs a real footprint (clergy split carves the liege's portion)
			custom_tooltip = {
				text = pam_split_too_small_tt
				scope:puppet_or_actor.liege ?= { split_clerical_region_size >= 3 }

			# The new region must not be larger than what the superior keeps (at most half of the region)
			custom_tooltip = {
				text = pam_split_larger_than_superior_tt
				scope:puppet_or_actor.liege ?= {
					split_clerical_region_size <= {
						value = capital_county.clerical_region_title.clerical_region_size
						divide = 2
					}
				}
			}

			scope:clerical_region_holder = {

			hidden_effect = {

replace_ecclesiastical_title_holder_interaction = {
	category = interaction_category_religion
	icon = icon_clergy_move_lands
	interface_priority = 45
	common_interaction = yes

	use_diplomatic_range = yes
	ai_maybe = no
	popup_on_receive = yes

	desc = replace_ecclesiastical_title_holder_interaction_desc
	target_type = title
	target_filter = recipient_domain_titles
	show_all_targets = no

	# Actor's court chaplain as the default replacement; must match their entry in populate_recipient_list below
	redirect = {
		scope:actor = {
			if = {
				limit = {
					cp:councillor_court_chaplain ?= {
						NOT = { any_held_title = { has_clerical_region = yes } }
						can_be_court_chaplain_trigger = { COURT_OWNER = scope:actor }
						trigger_if = {
							limit = { is_landed = yes }
							capital_county.clerical_region_title.holder ?= scope:recipient
						}
					}
				}
				cp:councillor_court_chaplain ?= {
					save_scope_as = secondary_recipient
				}
			}
		}
	}

	# Landed candidates must be based in the recipient's archdiocese, or their lands would drag the see into a foreign realm
	populate_recipient_list = {
		scope:actor = {
			if = {
				limit = {
					cp:councillor_court_chaplain ?= {
						NOT = { any_held_title = { has_clerical_region = yes } }
						can_be_court_chaplain_trigger = { COURT_OWNER = scope:actor }
						trigger_if = {
							limit = { is_landed = yes }
							capital_county.clerical_region_title.holder ?= scope:recipient
						}
					}
				}
				cp:councillor_court_chaplain ?= { add_to_list = characters }
			}
			every_courtier = {
				limit = {
					NOT = { any_held_title = { has_clerical_region = yes } }
					can_be_court_chaplain_trigger = { COURT_OWNER = scope:actor }
				}
				add_to_list = characters
			}
			every_close_family_member = {
				limit = {
					NOT = { any_held_title = { has_clerical_region = yes } }
					can_be_court_chaplain_trigger = { COURT_OWNER = scope:actor }
					trigger_if = {
						limit = { is_landed = yes }
						capital_county.clerical_region_title.holder ?= scope:recipient
					}
				}
				add_to_list = characters
			}
			every_vassal = {
				limit = {
					NOT = { any_held_title = { has_clerical_region = yes } }
					can_be_court_chaplain_trigger = { COURT_OWNER = scope:actor }
					government_has_flag = government_is_ecclesiastical
					trigger_if = {
						limit = { is_landed = yes }
						capital_county.clerical_region_title.holder ?= scope:recipient
					}
				}
				add_to_list = characters
			}
		}
	}

	# Church authority (HoF / Head of Rites), or a liege the faith's Clerical Appointment doctrine empowers
	is_available = {
		faith ?= { has_doctrine_parameter = has_clerical_regions }
		OR = {
			# A secular liege when the doctrine grants rulers the right (Temporal/Revocable, Conditional)
			AND = {
				faith ?= { has_doctrine_parameter = clerical_appointment_ruler_can_replace_ecc }
				government_is_religious_trigger = no
			}
			# A clergy liege under Conditional (which carries both rights)
			AND = {
				faith ?= {
					has_doctrine_parameter = clerical_appointment_hof_can_replace_ecc
					has_doctrine_parameter = clerical_appointment_ruler_can_replace_ecc
				}
				government_is_religious_trigger = yes
			}
			# HoF / Head of Rites - blocked only under Temporal, Revocable (the crown alone holds the right then)
			AND = {
				faith ?= {
					OR = {
						has_doctrine_parameter = clerical_appointment_hof_can_replace_ecc
						NOT = { has_doctrine_parameter = clerical_appointment_ruler_can_replace_ecc }
					}
				}
