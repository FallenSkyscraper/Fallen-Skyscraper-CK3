fs_start_challenge_status_interaction = {
	icon = icon_scheme_challenge_status
	interface_priority = 80
	category = interaction_category_hostile
	common_interaction = no
	send_name = START_SCHEME
	scheme = fs_seoul_challenge_status
	ignores_pending_interaction_block = yes

	ai_targets = { ai_recipients = peer_vassals max = 5 }
	ai_frequency_by_tier = {
		barony = 0
		county = 92
		duchy = 48
		kingdom = 24
		empire = 12
		hegemony = 0
	}

	is_shown = {
		scope:actor = { government_has_flag = government_is_seoul }
		scope:recipient != scope:actor
		scope:recipient = {
			government_has_flag = government_is_seoul
			top_liege = scope:actor.top_liege
		}
	}
	is_valid_showing_failures_only = {
		scope:recipient = { is_lowborn = no }
		scope:actor = {
			is_adult = yes
			is_imprisoned = no
			can_start_scheme = {
				type = fs_seoul_challenge_status
				target_character = scope:recipient
			}
		}
	}

	desc = fs_start_challenge_status_interaction_desc
	options_heading = schemes.t.agent_packages
	send_options_exclusive = yes
	send_option = {
		flag = agent_focus_balance
		current_description = start_challenge_status_interaction.tt.agent_focus_balance
	}
	send_option = {
		flag = agent_focus_success
		current_description = start_challenge_status_interaction.tt.agent_focus_success
	}
	send_option = {
		flag = agent_focus_speed
		current_description = start_challenge_status_interaction.tt.agent_focus_speed
	}

	on_accept = {
		scope:actor = {
			if = {
				limit = { scope:agent_focus_success ?= yes }
				begin_scheme_with_agents_effect = {
					SCHEME_TYPE = fs_seoul_challenge_status
					TARGET_TYPE = target_character
					TARGET_SCOPE = scope:recipient
					AGENT_1 = agent_proponent
					AGENT_2 = agent_intermediary
					AGENT_3 = agent_sycophant
					AGENT_4 = agent_political_socialite
					AGENT_5 = agent_bureaucrat
				}
			}
			else_if = {
				limit = { scope:agent_focus_speed ?= yes }
				begin_scheme_with_agents_effect = {
					SCHEME_TYPE = fs_seoul_challenge_status
					TARGET_TYPE = target_character
					TARGET_SCOPE = scope:recipient
					AGENT_1 = agent_proponent
					AGENT_2 = agent_intermediary
					AGENT_3 = agent_political_socialite
					AGENT_4 = agent_bureaucrat
					AGENT_5 = agent_bureaucrat
				}
			}
			else = {
				begin_scheme_with_agents_effect = {
					SCHEME_TYPE = fs_seoul_challenge_status
					TARGET_TYPE = target_character
					TARGET_SCOPE = scope:recipient
					AGENT_1 = agent_proponent
					AGENT_2 = agent_intermediary
					AGENT_3 = agent_political_socialite
					AGENT_4 = agent_bureaucrat
					AGENT_5 = agent_amanuensis
				}
			}
			custom_tooltip = challenge_status_success_actor_tt
		}
	}
	auto_accept = yes
	ai_potential = {
		government_has_flag = government_is_seoul
		ai_honor <= 20
	}
	ai_will_do = { base = 20 }
}

seoul_influence_candidacy_interaction = {
	icon = support_candidacy
	category = interaction_category_succession
	common_interaction = yes
	interface_priority = 100
	desc = seoul_influence_candidacy_interaction_desc
	target_type = title
	target_filter = recipient_de_jure_titles
	custom_character_sort = { candidate_score }

	populate_recipient_list = {
		scope:actor.top_liege = {
			every_vassal_or_below = {
				limit = {
					government_has_flag = government_is_seoul
					fs_seoul_is_appointment_valid_trigger = {
						TITLE = title:k_seoul
						CANDIDATE = this
						TOP_LIEGE = scope:actor.top_liege
					}
				}
				add_to_list = characters
			}
		}
	}
	can_be_picked_title = {
		trigger_if = {
			limit = {
				exists = scope:target
				exists = scope:secondary_recipient
				exists = scope:recipient
			}
			scope:target = {
				custom_tooltip = {
					text = fs_seoul_candidacy_title_tt
					has_title_law = seoul_acclamation_succession_law
				}
			}
			fs_seoul_is_appointment_valid_trigger = {
				TITLE = scope:target
				CANDIDATE = scope:secondary_recipient
				TOP_LIEGE = scope:recipient
			}
		}
	}
	has_valid_target_showing_failures_only = {
		exists = scope:target
		scope:target = {
			custom_tooltip = {
				text = fs_seoul_candidacy_title_tt
				has_title_law = seoul_acclamation_succession_law
			}
			exists = holder
			holder.top_liege = scope:recipient
		}
	}
	is_available = {
		is_ai = no
		government_has_flag = government_is_seoul
	}
	is_valid_showing_failures_only = {
		scope:actor = { government_has_flag = government_is_seoul }
		scope:secondary_recipient ?= {
			government_has_flag = government_is_seoul
			top_liege = scope:actor.top_liege
		}
	}
	redirect = {
		scope:recipient = {
			if = {
				limit = { this != scope:actor.top_liege }
				save_scope_as = secondary_recipient
				scope:actor.top_liege = { save_scope_as = recipient }
			}
		}
	}
	auto_accept = yes
	send_options_exclusive = yes
	send_option = {
		flag = seoul_major_influence_option
		localization = seoul_major_influence_option_desc
	}
	send_option = {
		flag = seoul_medium_influence_option
		localization = seoul_medium_influence_option_desc
		starts_enabled = { always = yes }
	}
	cost = {
		influence = {
			if = {
				limit = { scope:seoul_major_influence_option = yes }
				add = appointment_major_influence_cost
			}
			else = { add = appointment_medium_influence_cost }
		}
	}
	on_accept = {
		if = {
			limit = { scope:seoul_major_influence_option = yes }
			save_scope_value_as = { name = seoul_support_score value = appointment_score_major_value }
		}
		else = {
			save_scope_value_as = { name = seoul_support_score value = appointment_score_medium_value }
		}
		scope:target = {
			change_appointment_investment = {
				target = scope:secondary_recipient
				investor = scope:actor
				value = scope:seoul_support_score
			}
		}
		custom_tooltip = seoul_influence_candidacy_success_tt
	}
}

fs_seoul_purchase_truce_interaction = {
	interface_priority = 30
	category = interaction_category_diplomacy
	common_interaction = yes
	desc = fs_seoul_purchase_truce_interaction_desc
	icon = icon_retreat
	greeting = positive
	notification_text = fs_seoul_purchase_truce_notification
	popup_on_receive = yes
	ai_maybe = yes
	auto_accept = no

	is_available = {
		fs_seoul_realm_ruler_trigger = yes
		NOT = { has_title = title:k_seoul }
		trigger_if = {
			limit = { is_ai = yes }
			is_at_war = no
			is_physically_able = yes
			influence >= fs_seoul_purchase_truce_cost
		}
	}
	is_shown = { fs_seoul_purchase_truce_participants_trigger = yes }
	is_valid_showing_failures_only = { fs_seoul_purchase_truce_valid_trigger = yes }
	on_accept = {
		custom_tooltip = fs_seoul_purchase_truce_effects_tt
		hidden_effect = { fs_seoul_purchase_truce_accept_effect = yes }
	}
	on_decline = {
		scope:actor = {
			send_interface_toast = {
				type = event_toast_effect_neutral
				title = fs_seoul_purchase_truce_declined
				left_icon = scope:recipient
			}
		}
	}

	# Use the native medium-offer baseline and personality/strength judgments.
	# Gold balance, selectable sums and struggle participation do not apply.
	ai_accept = {
		base = 0
		modifier = {
			add = 25
			desc = fs_seoul_purchase_truce_influence_offer_reason
		}
		opinion_modifier = {
			who = scope:recipient
			opinion_target = scope:actor
			multiplier = 0.5
			desc = AI_OPINION_REASON
		}
		modifier = {
			add = { value = ai_greed multiply = 2 }
			ai_greed >= 1
			desc = AI_VALUE_MODIFIER_GREED
		}
		modifier = {
			add = { value = ai_boldness multiply = -0.5 }
			NOT = { ai_boldness = 0 }
			desc = AI_VALUE_MODIFIER_BOLDNESS
		}
		modifier = {
			add = -50
			has_trait = ambitious
			desc = PT_AI_LIKES_FREEDOM_TO_WAR_REASON
		}
		modifier = {
			add = 25
			faith = { has_doctrine_parameter = pacifist }
			desc = PT_AI_IS_PACIFIST_FAITH
		}
		modifier = {
			add = 25
			culture = { has_cultural_parameter = reduced_cost_for_conversion_to_pacifist_faith }
			desc = PT_AI_IS_PACIFIST_CULTURE
		}
		modifier = {
			add = -30
			scope:actor.max_military_strength <= scope:recipient.purchase_truce_interaction_recipient_far_weaker_value
			desc = PT_AI_RECIPIENT_IS_MUCH_STRONGER
		}
		modifier = {
			add = -10
			scope:actor.max_military_strength > scope:recipient.purchase_truce_interaction_recipient_far_weaker_value
			scope:actor.max_military_strength <= scope:recipient.purchase_truce_interaction_recipient_somewhat_weaker_value
			desc = PT_AI_RECIPIENT_IS_SOMEWHAT_STRONGER
		}
		modifier = {
			add = 10
			scope:actor.max_military_strength > scope:recipient.purchase_truce_interaction_recipient_somewhat_weaker_value
			scope:actor.max_military_strength <= scope:recipient.purchase_truce_interaction_recipient_somewhat_stronger_value
			desc = PT_AI_RECIPIENT_IS_ROUGHLY_EQUAL
		}
		modifier = {
			add = 25
			scope:actor.max_military_strength > scope:recipient.purchase_truce_interaction_recipient_somewhat_stronger_value
			scope:actor.max_military_strength <= scope:recipient.purchase_truce_interaction_recipient_far_stronger_value
			desc = PT_AI_RECIPIENT_IS_SOMEWHAT_WEAKER
		}
		modifier = {
			add = 50
			scope:actor.max_military_strength > scope:recipient.purchase_truce_interaction_recipient_far_stronger_value
			desc = PT_AI_RECIPIENT_IS_MUCH_WEAKER
		}
	}
	ai_target_quick_trigger = { adult = yes }
	ai_targets = { ai_recipients = peer_vassals max = 10 }
	ai_frequency_by_tier = {
		barony = 0
		county = 60
		duchy = 60
		kingdom = 0
		empire = 0
		hegemony = 0
	}
	ai_will_do = {
		base = 0
		modifier = {
			add = 25
			has_trait = craven
			NOT = { has_trait = ambitious }
		}
		modifier = {
			add = 25
			ai_boldness <= very_high_negative_ai_value
			NOT = { has_trait = ambitious }
		}
		modifier = { add = 25 faith = { has_doctrine_parameter = pacifist } }
		modifier = { add = 25 culture = { has_cultural_parameter = reduced_cost_for_conversion_to_pacifist_faith } }
		modifier = { add = -25 has_trait = ambitious }
		modifier = { add = -25 has_trait = wrathful }
		modifier = { add = -50 has_trait = arbitrary }
		modifier = {
			add = 50
			scope:recipient.max_military_strength >= scope:actor.purchase_truce_interaction_recipient_somewhat_stronger_value
		}
		modifier = {
			add = 50
			scope:recipient.max_military_strength >= scope:actor.purchase_truce_interaction_recipient_far_stronger_value
		}
		modifier = {
			factor = 0
			scope:recipient = { NOT = { has_any_cb_on = scope:actor } }
		}
		modifier = {
			factor = 0
			OR = {
				health <= poor_health
				scope:recipient = { health <= poor_health }
			}
		}
	}
}
