# Based on CK3 1.20.1.1 00_pam_clerical_elective.txt. Keep native sets intact.

theocratic_elective = {
	elector_vote_strength = {
		base = 1
	}

	electors = {
		character_sets = {
			theocratic_electors_clerical = {
				type = same_faith_clerical_electors

				# Not the HoF
				limit = {
					this != scope:holder
				}
			}
		}
	}

	candidate_score = {
		base = 0
		######################	Elector self-voting pattern	##########################
		elector_self_voting_pattern_theocratic_elective_modifier = yes

		##########################	Theocratic elective patterns (misc)	##########################
		elector_voting_pattern_theocratic_elective_modifier = yes
		# A score penalty, not a candidacy ban. Other faiths retain native behavior.
		modifier = {
			desc = fs_constituency_presidential_offender_penalty_tt
			scope:title = title:k_koreanism
			scope:candidate = { has_trait = fs_election_offender }
			add = fs_constituency_presidential_offender_penalty
		}
	}

	candidates = {
		# General priority is added to every native pious/populist/powerful set.
		priority_value = { value = fs_constituency_presidential_offender_penalty_value }
		limit = {
			pam_papabile_barred_by_lay_clergy_trigger = no
			# Without the DLC a player cannot play a theocracy, so winning would end their game
			NAND = {
				is_ai = no
				has_pam_dlc_trigger = no
			}
		}

		character_sets = {
			theocratic_elective_pious = {
				type = same_faith_clerical_electors

				# We exclude campaigners outright, since a score penalty could be outweighed by high piety
				limit = {
					NOR = {
						exists = var:pam_campaign_as_powerful
						exists = var:pam_campaign_as_populist
						exists = var:pam_refuses_papacy
					}
				}

				# "The Pious" - Holiness of life
				priority_value = {
					value = pam_papal_pious_priority_value
				}
			}

			theocratic_elective_populist = {
				type = same_faith_clerical_electors

				limit = {
					NOR = {
						exists = var:pam_campaign_as_powerful
						exists = var:pam_refuses_papacy
					}
				}

				# "The Populist" - The people's champion
				priority_value = {
					value = pam_papal_populist_priority_value
				}
			}

			theocratic_elective_powerful = {
				type = same_faith_clerical_electors

				limit = {
					NOR = {
						exists = var:pam_campaign_as_populist
						exists = var:pam_refuses_papacy
					}
				}

				# "The Powerful" - Worldly might
				priority_value = {
					value = pam_papal_powerful_priority_value
				}
			}
		}

		candidate_policy = one_candidate_per_set
	}
}
