# Character scope. Resolve a permanent home, never the current travel location.
fs_constituency_court_home_in_region_trigger = {
	trigger_if = {
		limit = { is_landed = yes }
		capital_province = { geographical_region = $REGION$ }
	}
	trigger_else_if = {
		limit = { exists = domicile.domicile_location }
		domicile.domicile_location = { geographical_region = $REGION$ }
	}
	trigger_else = {
		exists = capital_province
		capital_province = { geographical_region = $REGION$ }
	}
}

fs_constituency_home_in_region_trigger = {
	trigger_if = {
		limit = { is_landed = yes }

	}
	trigger_else_if = {
		limit = { exists = domicile.domicile_location }
		domicile.domicile_location = { geographical_region = $REGION$ }
	}
	trigger_else_if = {
		limit = {
			exists = court_owner
			this != court_owner
		}

	}
	trigger_else = {
		exists = capital_province
		capital_province = { geographical_region = $REGION$ }
	}
}

# Character scope. The chosen option remains valid after reserving its own seat.
# A new planner or a competing activity still cannot select that reservation.
fs_constituency_option_valid_trigger = {
	fs_constituency_organizer_trigger = yes
	$SEAT$ = {
		fs_koreanism_constituency_parliament_seat_trigger = yes
		fs_koreanism_vacant_parliament_seat_trigger = yes
		OR = {
			NOT = { exists = var:fs_constituency_election_activity }
			AND = {
				exists = scope:activity
				var:fs_constituency_election_activity ?= scope:activity
				scope:activity ?= {
					var:fs_constituency_seat = $SEAT$
					NOT = { has_variable = fs_constituency_resolved }
				}
			}
		}
	}
	OR = {
		has_title = title:k_koreanism
		fs_constituency_resident_trigger = { SEAT = $SEAT$ }
	}
}

fs_constituency_has_hostable_seat_trigger = {

		prev = {
			OR = {
				has_title = title:k_koreanism
				fs_constituency_resident_trigger = { SEAT = prev }
			}
		}
	}
}

# Activity scope. Native creation can validate before selected-option on_start runs.
# Only the short creation window may omit reservation state. Once a seat is bound,
# strict ownership applies, including to failed reservations and older saved activities.
fs_constituency_activity_valid_trigger = {
	NOT = { has_variable = fs_constituency_resolved }
	trigger_if = {
		limit = { has_variable = fs_constituency_seat }
		fs_constituency_running_activity_trigger = yes
	}
	trigger_else = {
		scope:host = { fs_constituency_has_hostable_seat_trigger = yes }
	}
}

# Activity scope. Retain authority while travelling; do not test the host's current location.
fs_constituency_running_activity_trigger = {
	exists = var:fs_constituency_seat
	NOT = { has_variable = fs_constituency_resolved }
	var:fs_constituency_seat = {
		fs_koreanism_vacant_parliament_seat_trigger = yes
		var:fs_constituency_election_activity = scope:activity
	}
	scope:host = {
		fs_constituency_organizer_trigger = yes
		OR = {
			has_title = title:k_koreanism
			fs_constituency_resident_trigger = { SEAT = scope:activity.var:fs_constituency_seat }
		}
	}
}

# Character scope. ACTIVITY must be a named scope, not 'this' or 'prev'.
fs_constituency_registered_candidate_trigger = {
	var:fs_constituency_candidate_activity = $ACTIVITY$
	NOT = { has_variable = fs_constituency_disqualified }
	fs_constituency_candidate_for_seat_trigger = { SEAT = $ACTIVITY$.var:fs_constituency_seat }
	has_activity_intent = fs_constituency_candidate_intent
	involved_activity = $ACTIVITY$
	# Presence is distinct from residence: travel must not count as participation.
	location = $ACTIVITY$.activity_location
}

# Delayed events and open multiplayer windows must still belong to this exact round.
# Character event scope. Stale observer windows cannot grant XP, piety or stress relief.
fs_constituency_current_observer_event_trigger = {
	exists = scope:fs_constituency_event_activity
	is_alive = yes
	involved_activity = scope:fs_constituency_event_activity
	location = scope:fs_constituency_event_activity.activity_location
	NOT = { fs_constituency_registered_candidate_trigger = { ACTIVITY = scope:fs_constituency_event_activity } }
	scope:fs_constituency_event_activity = {
		fs_constituency_running_activity_trigger = yes
		var:fs_constituency_round = scope:fs_constituency_observer_round
		trigger_if = {
			limit = { has_variable = fs_constituency_closed_round }
			var:fs_constituency_closed_round < var:fs_constituency_round
		}
		trigger_if = {
			limit = { var:fs_constituency_round = 1 }
			NOT = { is_target_in_variable_list = { name = fs_constituency_observer_responses_1 target = root } }
		}
		trigger_else_if = {
			limit = { var:fs_constituency_round = 2 }
			NOT = { is_target_in_variable_list = { name = fs_constituency_observer_responses_2 target = root } }
		}
		trigger_else = {
			NOT = { is_target_in_variable_list = { name = fs_constituency_observer_responses_3 target = root } }
		}
	}
}
