fs_prk_election_is_idle_trigger = {
	NOT = { has_global_variable = fs_prk_election_phase }
}

fs_prk_election_is_voting_trigger = {
	has_global_variable = fs_prk_election_phase
	global_var:fs_prk_election_phase = 1
}

fs_prk_election_is_finalizing_trigger = {
	has_global_variable = fs_prk_election_phase
	global_var:fs_prk_election_phase = 2
}

# The underlying nomination window: the final full year before a regular
# election, or the 90-day nomination period for a special election.
fs_prk_election_campaign_window_is_open_trigger = {
	exists = title:k_prk.holder
	fs_prk_election_is_idle_trigger = yes
	OR = {
		has_global_variable = fs_prk_special_election_pending
		AND = {
			has_global_variable = election_term
			global_var:election_term = 0
		}
	}
}

# A round token is kept separately from the calendar window. Closing a round
# invalidates event windows that a multiplayer client may still have open.
fs_prk_election_campaign_is_open_trigger = {
	has_global_variable = fs_prk_campaign_round_active
	fs_prk_election_campaign_window_is_open_trigger = yes
}

fs_prk_campaign_event_belongs_to_current_round_trigger = {
	fs_prk_election_campaign_is_open_trigger = yes
	has_global_variable = fs_prk_campaign_generation
	exists = scope:fs_prk_campaign_event_generation
	has_variable = fs_prk_campaign_generation
	scope:fs_prk_campaign_event_generation = global_var:fs_prk_campaign_generation
	var:fs_prk_campaign_generation = scope:fs_prk_campaign_event_generation
}

fs_prk_campaign_event_can_be_used_trigger = {
	fs_prk_campaign_event_belongs_to_current_round_trigger = yes
	has_variable = fs_prk_candidate
	fs_prk_valid_election_candidate_trigger = yes
}

fs_prk_valid_election_candidate_trigger = {
	is_alive = yes
	OR = {
		has_title = title:k_prk
		AND = {
			is_adult = yes
			is_imprisoned = no
			top_liege = { has_title = title:k_prk }
		}
	}
}

# Eligibility for voluntarily registering as a candidate. This is deliberately
# stricter than the counting-stage trigger, which always retains the incumbent
# as a safety fallback.
fs_prk_can_register_election_candidate_trigger = {
	fs_prk_election_is_idle_trigger = yes
	is_alive = yes
	is_adult = yes
	is_imprisoned = no
	OR = {
		has_title = title:k_prk
		top_liege = { has_title = title:k_prk }
	}
	NOT = { has_variable = fs_prk_candidate }
}

# The candidate cap exists only to prevent AI candidate proliferation.
fs_prk_ai_candidate_slot_available_trigger = {
	NOT = {
		any_living_character = {
			has_variable = fs_prk_candidate
			fs_prk_valid_election_candidate_trigger = yes
			count >= 9
		}
	}
}
