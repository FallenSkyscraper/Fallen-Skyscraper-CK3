# Doctrine scope. These categories also define independent AI topic cooldowns.
fs_legislation_religion_policy_doctrine_trigger = {
	OR = {
		this = doctrine:religion_policy_ultranationalism
		this = doctrine:religion_policy_state_atheism
		this = doctrine:religion_policy_laicite
		this = doctrine:religion_policy_freedom_of_religion
		this = doctrine:religion_policy_pluralism
	}
}

fs_legislation_tolerance_doctrine_trigger = {
	OR = {
		this = doctrine:doctrine_pluralism_fundamentalist
		this = doctrine:doctrine_pluralism_righteous
		this = doctrine:doctrine_pluralism_pluralistic
	}
}

# Doctrine scope. Parliament may change these fields, never party tenets.
fs_legislation_allowed_doctrine_trigger = {
	OR = {
		fs_legislation_religion_policy_doctrine_trigger = yes
		fs_legislation_tolerance_doctrine_trigger = yes
	}
}

# Rite scope. Evaluate the resulting policy/tolerance pair without changing either here.
fs_legislation_rite_doctrine_compatible_trigger = {
	trigger_if = {
		limit = { $DOCTRINE$ = doctrine:religion_policy_ultranationalism }
		rite_has_doctrine = doctrine_pluralism_fundamentalist
	}
	trigger_else_if = {
		limit = { $DOCTRINE$ = doctrine:religion_policy_state_atheism }
		NOT = { rite_has_doctrine = doctrine_pluralism_pluralistic }
	}
	trigger_else_if = {
		limit = { $DOCTRINE$ = doctrine:religion_policy_freedom_of_religion }
		NOT = { rite_has_doctrine = doctrine_pluralism_fundamentalist }
	}
	trigger_else_if = {
		limit = { $DOCTRINE$ = doctrine:religion_policy_pluralism }
		rite_has_doctrine = doctrine_pluralism_pluralistic
	}
	trigger_else_if = {
		limit = { $DOCTRINE$ = doctrine:doctrine_pluralism_fundamentalist }
		NOR = {
			rite_has_doctrine = religion_policy_freedom_of_religion
			rite_has_doctrine = religion_policy_pluralism
		}
	}
	trigger_else_if = {
		limit = { $DOCTRINE$ = doctrine:doctrine_pluralism_righteous }
		NOR = {
			rite_has_doctrine = religion_policy_ultranationalism
			rite_has_doctrine = religion_policy_pluralism
		}
	}
	trigger_else_if = {
		limit = { $DOCTRINE$ = doctrine:doctrine_pluralism_pluralistic }
		NOR = {
			rite_has_doctrine = religion_policy_ultranationalism
			rite_has_doctrine = religion_policy_state_atheism
		}
	}
	trigger_else = {
		$DOCTRINE$ = doctrine:religion_policy_laicite
	}
}

# Character scope; scope:doctrine is supplied by the proposal or enactment caller.
# Identity/no-op checks are separate so UI can explain incompatibility independently.
fs_legislation_doctrine_change_valid_trigger = {
	exists = scope:doctrine
	fs_legislation_doctrine_candidate_valid_trigger = { DOCTRINE = scope:doctrine }
}

# Character scope. Shared by public validation and AI candidate filtering before selection.
fs_legislation_doctrine_candidate_valid_trigger = {
	faith = faith:koreanism_faith
	$DOCTRINE$ = { fs_legislation_allowed_doctrine_trigger = yes }
	faith:koreanism_faith = {
		NOT = {
			any_faith_rite = {
				NOT = {
					fs_legislation_rite_doctrine_compatible_trigger = { DOCTRINE = $DOCTRINE$ }
				}
			}
		}
	}
}

# Character scope; the GUI and effects share costs and complete payload validation.
fs_legislation_doctrine_proposal_valid_trigger = {

# AI proposals need a 20-point improvement after the shared amendment resistance.
# PREFERENCE/THRESHOLD are character script values from the central preference table.
fs_legislation_ai_doctrine_proposal_trigger = {
	fs_legislation_ai_law_proposal_trigger = yes
	fs_legislation_doctrine_candidate_valid_trigger = { DOCTRINE = $DOCTRINE$ }
	$PREFERENCE$ >= $THRESHOLD$
	title:k_koreanism = {
		NOT = { has_variable = $TOPIC_COOLDOWN$ }
		fs_legislation_doctrine_change_needed_trigger = { DOCTRINE = $DOCTRINE$ }
	}
}

fs_legislation_ai_dual_office_proposal_trigger = {
	fs_legislation_ai_law_proposal_trigger = yes
	fs_legislation_dual_office_proposal_valid_trigger = { POLICY = $POLICY$ }
	$PREFERENCE$ >= fs_legislation_ai_dual_office_proposal_threshold_value
	title:k_koreanism = { NOT = { has_variable = fs_legislation_ai_dual_office_cooldown } }
}

# Title scope. A law is meaningful if even one rite still needs the proposed doctrine.
fs_legislation_doctrine_change_needed_trigger = {
	this = title:k_koreanism
	$DOCTRINE$ = { fs_legislation_allowed_doctrine_trigger = yes }
	faith:koreanism_faith = {
		OR = {
			NOT = { has_doctrine = $DOCTRINE$ }
			any_faith_rite = { NOT = { rite_has_doctrine = $DOCTRINE$ } }
		}
	}
}

# Title scope. Missing policy values retain the permissive pre-law behavior.
# Comparing only the two supported flags also rejects malformed proposal payloads.
fs_legislation_dual_office_policy_change_needed_trigger = {
	this = title:k_koreanism
	OR = {
		AND = {
			$POLICY$ = flag:allowed
			fs_koreanism_parliament_dual_office_forbidden_trigger = yes
		}
		AND = {
			$POLICY$ = flag:forbidden
			NOT = { fs_koreanism_parliament_dual_office_forbidden_trigger = yes }
		}
	}
}

# Character scope. The GUI and public proposal effect share the same authority guard.
fs_legislation_dual_office_proposal_valid_trigger = {

	title:k_koreanism = {
		fs_legislation_dual_office_policy_change_needed_trigger = { POLICY = $POLICY$ }
	}
}
