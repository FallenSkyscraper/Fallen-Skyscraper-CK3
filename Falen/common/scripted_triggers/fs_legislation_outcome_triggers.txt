# Doctrine scope. Parliament may change these fields, never party tenets.

		fs_legislation_tolerance_doctrine_trigger = yes
	}
}

# Rite scope. Evaluate the resulting policy/tolerance pair without changing either here.
fs_legislation_rite_doctrine_compatible_trigger = {
	trigger_if = {
		limit = { $DOCTRINE$ = doctrine:religion_policy_ultranationalism }
		rite_has_doctrine = doctrine_pluralism_fundamentalist
	}
	trigger_else_if = {
		limit = { $DOCTRINE$ = doctrine:religion_policy_state_atheism }
		NOT = { rite_has_doctrine = doctrine_pluralism_pluralistic }
	}
	trigger_else_if = {
		limit = { $DOCTRINE$ = doctrine:religion_policy_freedom_of_religion }
		NOT = { rite_has_doctrine = doctrine_pluralism_fundamentalist }
	}
	trigger_else_if = {
		limit = { $DOCTRINE$ = doctrine:religion_policy_pluralism }
		rite_has_doctrine = doctrine_pluralism_pluralistic
	}
	trigger_else_if = {
		limit = { $DOCTRINE$ = doctrine:doctrine_pluralism_fundamentalist }
		NOR = {
			rite_has_doctrine = religion_policy_freedom_of_religion
			rite_has_doctrine = religion_policy_pluralism
		}
	}
	trigger_else_if = {
		limit = { $DOCTRINE$ = doctrine:doctrine_pluralism_righteous }
		NOR = {
			rite_has_doctrine = religion_policy_ultranationalism
			rite_has_doctrine = religion_policy_pluralism
		}
	}
	trigger_else_if = {
		limit = { $DOCTRINE$ = doctrine:doctrine_pluralism_pluralistic }
		NOR = {
			rite_has_doctrine = religion_policy_ultranationalism
			rite_has_doctrine = religion_policy_state_atheism
		}
	}
	trigger_else = {
		$DOCTRINE$ = doctrine:religion_policy_laicite
	}
}

# Character scope. The draft is temporary; submission still uses public proposal guards.
fs_legislation_draft_navigation_trigger = {
	has_variable = fs_legislation_draft_open
	fs_legislation_can_propose_trigger = yes
}

fs_legislation_draft_tenet_valid_trigger = {
	trigger_if = {
		limit = { var:fs_legislation_draft_kind = flag:tenet_permitted }
		fs_legislation_tenet_proposal_valid_trigger = { TENET = $TENET$ STATUS = permitted }
	}
	trigger_else_if = {
		limit = { var:fs_legislation_draft_kind = flag:tenet_prohibited }
		fs_legislation_tenet_proposal_valid_trigger = { TENET = $TENET$ STATUS = prohibited }
	}
	trigger_else = { always = no }
}

fs_legislation_draft_valid_trigger = {
	fs_legislation_draft_navigation_trigger = yes
	var:fs_legislation_draft_step = flag:confirm
	trigger_if = {
		limit = { var:fs_legislation_draft_kind = flag:doctrine }
		exists = var:fs_legislation_draft_doctrine
		var:fs_legislation_draft_doctrine = { save_temporary_scope_as = doctrine }
		fs_legislation_doctrine_proposal_valid_trigger = yes
	}
	trigger_else_if = {
		limit = { var:fs_legislation_draft_kind = flag:dual_office }
		OR = {
			AND = { var:fs_legislation_draft_policy = flag:allowed fs_legislation_dual_office_proposal_valid_trigger = { POLICY = flag:allowed } }
			AND = { var:fs_legislation_draft_policy = flag:forbidden fs_legislation_dual_office_proposal_valid_trigger = { POLICY = flag:forbidden } }
		}
	}
	trigger_else_if = {
		limit = { OR = { var:fs_legislation_draft_kind = flag:tenet_permitted var:fs_legislation_draft_kind = flag:tenet_prohibited } }
		exists = var:fs_legislation_draft_tenet
		var:fs_legislation_draft_tenet = { save_temporary_scope_as = fs_legislation_draft_selected_tenet }
		fs_legislation_draft_tenet_valid_trigger = { TENET = scope:fs_legislation_draft_selected_tenet }
	}
	trigger_else_if = {
		limit = { var:fs_legislation_draft_kind = flag:heresy }
		exists = var:fs_legislation_draft_rite
		var:fs_legislation_draft_rite = { save_temporary_scope_as = fs_legislation_draft_selected_rite }
		fs_legislation_heresy_proposal_valid_trigger = { RITE = scope:fs_legislation_draft_selected_rite }
	}
	trigger_else_if = {
		limit = { var:fs_legislation_draft_kind = flag:impeachment }
		NOT = { has_title = title:k_koreanism }
		exists = var:fs_legislation_draft_target
		var:fs_legislation_draft_target = title:k_koreanism.holder
	}
	trigger_else_if = {
		limit = { var:fs_legislation_draft_kind = flag:lift_martial_law }
		title:k_koreanism = { has_variable = fs_legislation_martial_law }
	}
	trigger_else_if = {
		limit = { OR = { var:fs_legislation_draft_kind = flag:censure var:fs_legislation_draft_kind = flag:expulsion } }
		exists = var:fs_legislation_draft_target
		this != var:fs_legislation_draft_target
		var:fs_legislation_draft_target = { fs_legislation_member_trigger = yes }
	}
	trigger_else = { always = no }
}
