# Classify by title identity: Sejong shares its capital with proportional seats.

	OR = {
		this = title:d_cd_prop_1
		this = title:d_cd_prop_2
		this = title:d_cd_prop_3

fs_koreanism_constituency_parliament_seat_trigger = {
	fs_koreanism_parliament_seat_trigger = yes
	NOT = { fs_koreanism_proportional_parliament_seat_trigger = yes }
}

# Only unreserved proportional vacancies can be filled by appointment or petition.
fs_koreanism_appointable_parliament_seat_trigger = {
	fs_koreanism_proportional_parliament_seat_trigger = yes
	fs_koreanism_vacant_parliament_seat_trigger = yes
	NOT = { exists = var:fs_constituency_election_activity }
}

# Constituency elections check their own vacancies independently of appointments.
fs_constituency_open_seat_trigger = {
	fs_koreanism_constituency_parliament_seat_trigger = yes
	fs_koreanism_vacant_parliament_seat_trigger = yes
	NOT = { exists = var:fs_constituency_election_activity }
}

# Character scope. Lay candidates and all Koreanism rites are eligible.
fs_koreanism_parliament_candidate_trigger = {
	faith = faith:koreanism_faith
	is_available_quick = {
		alive = yes
		adult = yes
		incapable = no
		imprisoned = no
		hostage = no
	}
	NOR = {
		has_title = title:k_koreanism
		exists = clerical_elector_title
		any_held_title = { fs_koreanism_parliament_seat_trigger = yes }
	}
}

# Character scope. Only the current president has parliamentary authority.
fs_koreanism_parliament_president_trigger = {
	faith = faith:koreanism_faith
	has_title = title:k_koreanism
	is_available_quick = {
		alive = yes
		adult = yes
		incapable = no
		imprisoned = no
		hostage = no
	}
}

# Character scope. Constituency vacancies never enable appointments or petitions.
fs_koreanism_has_proportional_vacancy_trigger = {
	any_in_global_list = {
		variable = fs_koreanism_parliament_seats
		fs_koreanism_appointable_parliament_seat_trigger = yes
	}
}

# Temporary policy gate; keep involuntary vacancies and election cleanup independent.
fs_koreanism_parliament_dismissal_enabled_trigger = {
	always = no
}

# Interaction scope. Requests are made by the candidate to the current president.
fs_koreanism_parliament_request_valid_trigger = {
	scope:puppet_or_actor = scope:actor
	custom_tooltip = {
		text = fs_parliament_candidate_tt
		scope:actor = { fs_koreanism_parliament_appointment_candidate_trigger = yes }
	}
	scope:recipient = { fs_koreanism_parliament_president_trigger = yes }
	scope:actor = {
		this != scope:recipient
		NOT = { is_at_war_with = scope:recipient }
	}
	custom_tooltip = {
		text = fs_parliament_vacancy_tt
		scope:actor = { fs_koreanism_has_proportional_vacancy_trigger = yes }
	}
}

# Any scope. Older saves without an explicit policy retain the existing permission.
fs_koreanism_parliament_dual_office_forbidden_trigger = {
	title:k_koreanism = { var:fs_parliament_dual_office_policy = flag:forbidden }
}

# Character scope. Constitutional and Koreanism regional-office land is public property.
# Other landed governments are covered, including clan, tribal and administrative rulers.
fs_koreanism_parliament_secular_lord_trigger = {
	is_landed = yes
	NOR = {
		has_government = koreanism_government
		AND = {
			faith = faith:koreanism_faith
			any_held_title = { has_clerical_region = yes }
		}
	}
}

# Character scope. Elections keep the broader candidate trigger so lords may still stand.
fs_koreanism_parliament_appointment_candidate_trigger = {
	fs_koreanism_parliament_candidate_trigger = yes
	custom_tooltip = {
		text = fs_parliament_dual_office_appointment_allowed_tt
		NAND = {
			fs_koreanism_parliament_dual_office_forbidden_trigger = yes
			fs_koreanism_parliament_secular_lord_trigger = yes
		}
	}
}

# Title scope. Exclude titles without claims, not offices barred only from buying claims.

	is_noble_family_title = no
	is_mercenary_company = no
	is_holy_order = no
	NOT = { has_variable = adventurer_creation_reason }
}
