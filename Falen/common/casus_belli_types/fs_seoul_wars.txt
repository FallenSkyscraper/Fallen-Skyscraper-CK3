seoul_recovery_cb = {
	group = seoul_recovery
	icon = de_jure_cb
	combine_into_one = yes
	should_show_war_goal_subview = yes
	mutually_exclusive_titles = { always = yes }

	allowed_for_character = {
		scope:attacker = {
			is_landed = yes
			government_has_flag = government_is_seoul
			top_liege ?= { has_title = title:k_seoul }
		}
	}
	allowed_against_character = {
		scope:attacker.top_liege != scope:defender.top_liege
	}

	target_titles = all
	target_title_tier = county
	ignore_effect = change_title_holder

	valid_to_start = {
		scope:target = {
			tier = tier_county
			holder = { target_is_same_character_or_below = scope:defender }
			title:k_seoul = {
				any_de_jure_county = { this = scope:target }
			}
		}
	}

	should_invalidate = {
		OR = {
			scope:attacker = { NOT = { government_has_flag = government_is_seoul } }
			scope:attacker.top_liege = scope:defender.top_liege
			scope:target = {
				OR = {
					NOT = { holder = { target_is_same_character_or_below = scope:defender } }
					title:k_seoul = {
						NOT = { any_de_jure_county = { this = scope:target } }
					}
				}
			}
		}
	}

	on_invalidated_desc = msg_county_conquest_war_invalidated_message
	cost = { prestige = { value = 75 multiply = common_cb_prestige_cost_multiplier } }
	on_declaration = { on_declared_war = yes }

	on_victory_desc = seoul_recovery_cb_victory_desc
	on_victory = {
		scope:attacker = { show_pow_release_message_effect = yes }
		create_title_and_vassal_change = {
			type = conquest
			save_scope_as = change
			add_claim_on_loss = yes
		}
		every_in_list = {
			list = target_titles
			change_title_holder = {
				holder = scope:attacker
				change = scope:change
			}
		}
		resolve_title_and_vassal_change = scope:change
		add_truce_attacker_victory_effect = yes
	}

	on_white_peace_desc = seoul_recovery_cb_white_peace_desc
	on_white_peace = {
		scope:attacker = { show_pow_release_message_effect = yes }
		add_truce_white_peace_effect = yes
	}

	on_defeat_desc = seoul_recovery_cb_defeat_desc
	on_defeat = {
		scope:attacker = { show_pow_release_message_effect = yes }
		add_truce_attacker_defeat_effect = yes
	}

	transfer_behavior = transfer
	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit
	attacker_allies_inherit = no
	defender_allies_inherit = yes
	war_name = SEOUL_RECOVERY_WAR_NAME
	war_name_base = SEOUL_RECOVERY_WAR_NAME_BASE
	cb_name = SEOUL_RECOVERY_CB_NAME
	interface_priority = 100
	use_de_jure_wargoal_only = yes
	attacker_wargoal_percentage = 0.95
}

seoul_influence_dispute_cb = {
	group = seoul_influence_dispute
	icon = humiliation_cb
	allow_hostages = no

	allowed_for_character = {
		scope:attacker = {
			is_landed = yes
			government_has_flag = government_is_seoul
			exists = liege
		}
	}
	allowed_against_character = {
		scope:defender = {
			is_landed = yes
			government_has_flag = government_is_seoul
			exists = liege
			liege = scope:attacker.liege
		}
		scope:defender != scope:attacker
	}

	ignore_effect = change_title_holder
	should_invalidate = {
		OR = {
			scope:attacker = { NOT = { government_has_flag = government_is_seoul } }
			scope:defender = { NOT = { government_has_flag = government_is_seoul } }
			scope:defender.liege != scope:attacker.liege
		}
	}
	on_invalidated_desc = msg_remove_humiliation_cb_invalidated_message
	cost = { prestige = { value = 75 multiply = common_cb_prestige_cost_multiplier } }
	on_declaration = { on_declared_war = yes }

	on_victory_desc = seoul_influence_dispute_cb_victory_desc
	on_victory = {
		scope:attacker = {
			show_pow_release_message_effect = yes
			change_influence = { value = 100 }
		}
		scope:defender = { change_influence = { value = -100 } }
		add_truce_attacker_victory_effect = yes
	}

	on_white_peace_desc = seoul_influence_dispute_cb_white_peace_desc
	on_white_peace = {
		scope:attacker = { show_pow_release_message_effect = yes }
		add_truce_white_peace_effect = yes
	}

	on_defeat_desc = seoul_influence_dispute_cb_defeat_desc
	on_defeat = {
		scope:attacker = {
			show_pow_release_message_effect = yes
			change_influence = { value = -100 }
		}
		scope:defender = { change_influence = { value = 100 } }
		add_truce_attacker_defeat_effect = yes
	}

	transfer_behavior = transfer
	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit
	attacker_allies_inherit = yes
	defender_allies_inherit = yes
	war_name = SEOUL_INFLUENCE_DISPUTE_WAR_NAME
	war_name_base = SEOUL_INFLUENCE_DISPUTE_WAR_NAME_BASE
	cb_name = SEOUL_INFLUENCE_DISPUTE_CB_NAME
	interface_priority = 95
	attacker_wargoal_percentage = 0.8
	max_defender_score_from_battles = 150
	max_attacker_score_from_battles = 150
	max_defender_score_from_occupation = 150
	max_attacker_score_from_occupation = 150
}
