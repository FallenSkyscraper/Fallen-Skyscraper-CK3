hanseatic_kontor_01 = {
	type_icon = old_buildings/capital_bureau.dds

	cost = { gold = trade_post_building_tier_1_cost_value }
	construction_time = trade_post_building_construction_time_value
	next_building = hanseatic_kontor_02

	can_construct_potential = {
		trigger_if = {
			limit = {
				exists = trade_post_district
			}
			trade_post_owner ?= {
				is_hanseatic_league_member_trigger = yes
			}
		}
	}

	is_enabled = {
		trigger_if = {
			limit = {
				exists = trade_post_district
			}
			trade_post_owner ?= {
				is_hanseatic_league_member_trigger = yes
			}
		}
	}

	character_modifier = {
		trade_competitiveness_add = 1
		trade_capacity_mult = 0.01
	}
	county_modifier = {
		prosperity_growth_factor = 0.05
		build_gold_cost = -0.05
		trade_category_all_sell_price_add = 0.0005
	}

	flag = hanseatic_kontor_flag

	ai_value = {
		base = 11
		ai_trade_post_new_building_modifier = yes
	}
}
