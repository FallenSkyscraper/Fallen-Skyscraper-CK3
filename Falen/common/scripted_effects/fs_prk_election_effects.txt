# Build the public electorate summary and the per-candidate AI support estimate.
# Exact ties are deterministic in the preview only; the real ballot still
# randomizes tied AI choices when voting starts.
fs_prk_refresh_election_overview_ui_effect = {
	save_temporary_scope_as = fs_prk_ui_owner
	clear_variable_list = fs_prk_ui_electors
	set_variable = { name = fs_prk_ui_elector_count value = 0 }
	set_variable = { name = fs_prk_ui_total_electoral_votes value = 0 }
	set_variable = { name = fs_prk_ui_human_electoral_votes value = 0 }
	set_variable = { name = fs_prk_ui_my_electoral_votes value = 0 }

	# GUI variable lists are display models and cannot be iterated as script lists.
	# Build dedicated temporary lists for all calculations in this refresh.
	if = {
		limit = {
			OR = {
				fs_prk_election_is_voting_trigger = yes
				fs_prk_election_is_finalizing_trigger = yes
			}
		}
		every_in_global_list = {
			variable = fs_prk_election_candidates
			limit = { fs_prk_valid_election_candidate_trigger = yes }
			add_to_temporary_list = fs_prk_ui_calculation_candidates
		}
	}
	else = {
		every_living_character = {
			limit = {
				has_variable = fs_prk_candidate
				fs_prk_valid_election_candidate_trigger = yes
			}
			add_to_temporary_list = fs_prk_ui_calculation_candidates
		}
	}

	# During a ballot, show the frozen electorate. Otherwise preview the current realm.
	if = {
		limit = {
			OR = {
				fs_prk_election_is_voting_trigger = yes
				fs_prk_election_is_finalizing_trigger = yes
			}
		}
		every_in_global_list = {
			variable = fs_prk_election_electors
			save_temporary_scope_as = fs_prk_ui_elector
			add_to_temporary_list = fs_prk_ui_calculation_electors
			scope:fs_prk_ui_owner = {
				add_to_variable_list = {
					name = fs_prk_ui_electors
					target = scope:fs_prk_ui_elector
				}
			}
		}
	}
	else_if = {
		limit = { exists = title:k_prk.holder }
		title:k_prk.holder = {
			every_realm_county = {
				holder = {
					save_temporary_scope_as = fs_prk_ui_elector
					if = {
						limit = { NOT = { is_in_list = fs_prk_ui_calculation_electors } }
						add_to_temporary_list = fs_prk_ui_calculation_electors
						scope:fs_prk_ui_owner = {
							add_to_variable_list = {
								name = fs_prk_ui_electors
								target = scope:fs_prk_ui_elector
							}
						}
					}
				}
			}
		}
	}

	# Calculate the development-weighted totals used by the election.
	every_in_list = {
		list = fs_prk_ui_calculation_candidates
		set_variable = { name = fs_prk_ui_expected_ai_votes value = 0 }
		set_variable = { name = fs_prk_ui_electoral_votes value = 0 }
	}
	every_in_list = {
		list = fs_prk_ui_calculation_electors
		save_temporary_scope_as = fs_prk_ui_elector
		set_variable = { name = fs_prk_ui_electoral_votes value = 0 }
		if = {
			limit = {
				OR = {
					fs_prk_election_is_voting_trigger = yes
					fs_prk_election_is_finalizing_trigger = yes
				}
				has_variable = fs_prk_election_voting_power_snapshot
			}
			set_variable = {
				name = fs_prk_ui_electoral_votes
				value = var:fs_prk_election_voting_power_snapshot
			}
		}
		else = {
			every_held_title = {
				limit = {
					tier = tier_county
					development_level > 0
				}
				scope:fs_prk_ui_elector = {
					change_variable = {
						name = fs_prk_ui_electoral_votes
						add = prev.development_level
					}
				}
			}
		}

		scope:fs_prk_ui_owner = {
			change_variable = { name = fs_prk_ui_elector_count add = 1 }
			change_variable = {
				name = fs_prk_ui_total_electoral_votes
				add = scope:fs_prk_ui_elector.var:fs_prk_ui_electoral_votes
			}
		}
		if = {
			limit = { is_ai = no }
			scope:fs_prk_ui_owner = {
				change_variable = {
					name = fs_prk_ui_human_electoral_votes
					add = scope:fs_prk_ui_elector.var:fs_prk_ui_electoral_votes
				}
			}
		}
		if = {
			limit = { this = scope:fs_prk_ui_owner }
			scope:fs_prk_ui_owner = {
				set_variable = {
					name = fs_prk_ui_my_electoral_votes
					value = scope:fs_prk_ui_elector.var:fs_prk_ui_electoral_votes
				}
			}
		}
	}

	# Once voting has started, the AI ballot is frozen and can be shown exactly.
	if = {
		limit = {
			OR = {
				fs_prk_election_is_voting_trigger = yes
				fs_prk_election_is_finalizing_trigger = yes
			}
		}
		every_in_list = {
			list = fs_prk_ui_calculation_candidates
			set_variable = {
				name = fs_prk_ui_expected_ai_votes
				value = var:fs_prk_election_ai_votes
			}
		}
	}
	else = {
		# Preview every AI elector under the same explicit score as the real ballot.
		every_in_list = {
			list = fs_prk_ui_calculation_electors
			limit = { is_ai = yes }
			save_temporary_scope_as = fs_prk_elector
			scope:fs_prk_ui_owner = {
				every_in_list = {
					list = fs_prk_ui_calculation_candidates
					fs_prk_calculate_ai_election_score_effect = yes
				}
				ordered_in_list = {
					list = fs_prk_ui_calculation_candidates
					order_by = var:fs_prk_election_ai_score
					position = 0
					save_temporary_scope_as = fs_prk_ui_expected_candidate
				}
			}
			if = {
				limit = { exists = scope:fs_prk_ui_expected_candidate }
				scope:fs_prk_ui_expected_candidate = {
					change_variable = {
						name = fs_prk_ui_expected_ai_votes
						add = scope:fs_prk_elector.var:fs_prk_ui_electoral_votes
					}
				}
			}
			scope:fs_prk_ui_owner = {
				every_in_list = {
					list = fs_prk_ui_calculation_candidates
					remove_variable = fs_prk_election_ai_score
				}
			}
		}
	}
}

# Score the current candidate from the perspective of scope:fs_prk_elector.
fs_prk_calculate_ai_election_score_effect = {
	save_temporary_scope_as = fs_prk_scored_candidate
	set_variable = { name = fs_prk_election_ai_score value = 100 }

	# Personal preference.
	scope:fs_prk_elector = {
		save_opinion_value_as = {
			name = fs_prk_election_opinion
			target = scope:fs_prk_scored_candidate
		}
	}
	change_variable = {
		name = fs_prk_election_ai_score
		add = scope:fs_prk_election_opinion
	}
	if = {
		limit = { this = scope:fs_prk_elector }
		change_variable = { name = fs_prk_election_ai_score add = 1000 }
	}

	# Institutional continuity.
	if = {
		limit = { has_title = title:k_prk }
		change_variable = { name = fs_prk_election_ai_score add = 20 }
	}
	if = {
		limit = { is_diarch_of_target = title:k_prk.holder }
		change_variable = { name = fs_prk_election_ai_score add = 40 }
	}

	# Strong personal relationships are deliberately reinforced beyond opinion.
	if = {
		limit = {
			OR = {
				has_relation_best_friend = scope:fs_prk_elector
				has_relation_soulmate = scope:fs_prk_elector
			}
		}
		change_variable = { name = fs_prk_election_ai_score add = 50 }
	}
	else_if = {
		limit = {
			OR = {
				has_relation_friend = scope:fs_prk_elector
				has_relation_lover = scope:fs_prk_elector
			}
		}
		change_variable = { name = fs_prk_election_ai_score add = 25 }
	}
	if = {
		limit = { has_relation_nemesis = scope:fs_prk_elector }
		change_variable = { name = fs_prk_election_ai_score subtract = 80 }
	}
	else_if = {
		limit = { has_relation_rival = scope:fs_prk_elector }
		change_variable = { name = fs_prk_election_ai_score subtract = 40 }
	}
	if = {
		limit = { is_close_family_of = scope:fs_prk_elector }
		change_variable = { name = fs_prk_election_ai_score add = 20 }
	}
	if = {
		limit = {
			exists = house
			scope:fs_prk_elector = { exists = house }
			house = scope:fs_prk_elector.house
		}
		change_variable = { name = fs_prk_election_ai_score add = 10 }
	}

	# Religious and cultural affinity.
	if = {
		limit = { faith = scope:fs_prk_elector.faith }
		change_variable = { name = fs_prk_election_ai_score add = 15 }
	}
	else_if = {
		limit = {
			# Hostility is directional: the elector's faith judges the candidate's faith.
			scope:fs_prk_elector = {
				faith = {
					faith_hostility_level = {
						target = scope:fs_prk_scored_candidate.faith
						value = 1
					}
				}
			}
		}
		change_variable = { name = fs_prk_election_ai_score subtract = 10 }
	}
	else_if = {
		limit = {
			scope:fs_prk_elector = {
				faith = {
					faith_hostility_level = {
						target = scope:fs_prk_scored_candidate.faith
						value = 2
					}
				}
			}
		}
		change_variable = { name = fs_prk_election_ai_score subtract = 25 }
	}
	else_if = {
		limit = {
			scope:fs_prk_elector = {
				faith = {
					faith_hostility_level = {
						target = scope:fs_prk_scored_candidate.faith
						value = 3
					}
				}
			}
		}
		change_variable = { name = fs_prk_election_ai_score subtract = 50 }
	}
	if = {
		limit = { has_same_culture_as = scope:fs_prk_elector }
		change_variable = { name = fs_prk_election_ai_score add = 15 }
	}
	else_if = {
		limit = {
			culture = {
				has_same_culture_heritage = scope:fs_prk_elector.culture
			}
		}
		change_variable = { name = fs_prk_election_ai_score add = 5 }
	}

	# Administrative competence.
	change_variable = {
		name = fs_prk_election_ai_score
		add = diplomacy
	}
	change_variable = {
		name = fs_prk_election_ai_score
		add = { value = stewardship multiply = 0.75 }
	}
	change_variable = {
		name = fs_prk_election_ai_score
		add = { value = learning multiply = 0.5 }
	}
	change_variable = {
		name = fs_prk_election_ai_score
		add = { value = martial multiply = 0.25 }
	}
	change_variable = {
		name = fs_prk_election_ai_score
		add = { value = intrigue multiply = 0.25 }
	}

	# Public standing.
	change_variable = {
		name = fs_prk_election_ai_score
		add = { value = legitimacy_level multiply = 8 }
	}
	change_variable = {
		name = fs_prk_election_ai_score
		add = { value = prestige_level multiply = 4 }
	}

	# Traits with an obvious bearing on public office.
	if = {
		limit = { has_trait = diligent }
		change_variable = { name = fs_prk_election_ai_score add = 15 }
	}
	else_if = {
		limit = { has_trait = lazy }
		change_variable = { name = fs_prk_election_ai_score subtract = 15 }
	}
	if = {
		limit = { has_trait = just }
		change_variable = { name = fs_prk_election_ai_score add = 10 }
	}
	else_if = {
		limit = { has_trait = arbitrary }
		change_variable = { name = fs_prk_election_ai_score subtract = 10 }
	}
	if = {
		limit = { has_trait = honest }
		change_variable = { name = fs_prk_election_ai_score add = 5 }
	}
	else_if = {
		limit = { has_trait = deceitful }
		change_variable = { name = fs_prk_election_ai_score subtract = 5 }
	}
	if = {
		limit = { has_trait = incapable }
		change_variable = { name = fs_prk_election_ai_score subtract = 100 }
	}
}

fs_prk_try_schedule_election_finalization_effect = {
	if = {
		limit = {
			fs_prk_election_is_voting_trigger = yes
			global_var:fs_prk_election_pending_player_count <= 0
		}
		set_global_variable = { name = fs_prk_election_phase value = 2 }
		trigger_event = {
			id = joson_election.3
			days = 1
		}
	}
}

fs_prk_complete_player_vote_effect = {
	if = {
		limit = {
			fs_prk_election_is_voting_trigger = yes
			has_variable = fs_prk_election_vote_pending
		}
		remove_variable = fs_prk_election_vote_pending
		remove_variable = fs_prk_election_gui_toggle
		remove_variable = fs_prk_election_voting_power
		remove_variable = fs_prk_pending_vote_candidate
		remove_variable = fs_prk_pending_abstention
		remove_list_global_variable = {
			name = fs_prk_election_pending_players
			target = this
		}
		if = {
			limit = { global_var:fs_prk_election_pending_player_count > 0 }
			change_global_variable = {
				name = fs_prk_election_pending_player_count
				subtract = 1
			}
		}
		else = {
			set_global_variable = { name = fs_prk_election_pending_player_count value = 0 }
		}
		fs_prk_refresh_pending_player_ui_effect = yes
		fs_prk_try_schedule_election_finalization_effect = yes
	}
}
