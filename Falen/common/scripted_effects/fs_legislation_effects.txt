# Title scope throughout internal lifecycle effects. Public actions use character ROOT.
fs_legislation_initialize_effect = {
	if = {
		limit = { NOT = { has_variable = fs_legislation_serial } }
		set_variable = { name = fs_legislation_serial value = 0 }
	}
	# Existing saves and new games both start permissive until parliament changes the law.
	if = {
		limit = { NOT = { has_variable = fs_parliament_dual_office_policy } }
		set_variable = { name = fs_parliament_dual_office_policy value = flag:allowed }
	}
	if = {
		limit = {
			NOT = { any_in_global_list = { variable = fs_koreanism_parliament_seats always = yes } }
		}
		setup_cardinalate_list_effect = yes
	}
	if = {
		limit = { NOT = { has_variable = fs_legislation_clock_started } }
		set_variable = { name = fs_legislation_clock_started value = yes }
		trigger_event = { on_action = fs_legislation_heartbeat days = 1 }
	}
}

# Rebuild unique current members. A replacement receives a fresh ballot rather than
# inheriting the previous owner's selection. Multiple held seats still grant one vote.
fs_legislation_refresh_electorate_effect = {
	every_in_list = {
		variable = fs_legislation_voters
		remove_variable = fs_legislation_preserved_ballot
		if = {
			limit = {
				fs_legislation_member_trigger = yes
				exists = var:fs_legislation_ballot_seat
				var:fs_legislation_ballot_seat = {
					var:fs_legislation_ballot_serial = title:k_koreanism.var:fs_legislation_serial
					var:fs_legislation_ballot_holder = prev
					has_variable = fs_legislation_ballot_cast
				}
			}
			# Preserve this member's latest vote if their representative seat changes.
			set_variable = { name = fs_legislation_preserved_ballot value = var:fs_legislation_ballot_seat.var:fs_legislation_ballot }
		}
		remove_variable = fs_legislation_ballot_seat
	}
	clear_variable_list = fs_legislation_voters
	every_in_global_list = {
		variable = fs_koreanism_parliament_seats
		limit = {
			fs_koreanism_parliament_seat_trigger = yes
			exists = holder
			holder = { fs_legislation_member_trigger = yes }
		}
		save_scope_as = fs_legislation_seat
		holder = { save_scope_as = fs_legislation_elector }
		title:k_koreanism = {
			if = {
				limit = {
					NOT = { is_target_in_variable_list = { name = fs_legislation_voters target = scope:fs_legislation_elector } }
				}
				add_to_variable_list = { name = fs_legislation_voters target = scope:fs_legislation_elector }
				scope:fs_legislation_elector = {
					set_variable = { name = fs_legislation_ballot_seat value = scope:fs_legislation_seat }
				}
				scope:fs_legislation_seat = {
					set_variable = { name = fs_legislation_ballot_serial value = title:k_koreanism.var:fs_legislation_serial }
					set_variable = { name = fs_legislation_ballot_holder value = holder }
					set_variable = { name = fs_legislation_ballot value = 0 }
					remove_variable = fs_legislation_ballot_cast
					if = {
						limit = {
							scope:fs_legislation_elector = { has_variable = fs_legislation_preserved_ballot }
						}
						set_variable = { name = fs_legislation_ballot value = scope:fs_legislation_elector.var:fs_legislation_preserved_ballot }
						set_variable = { name = fs_legislation_ballot_cast value = yes }
					}
				}
				scope:fs_legislation_elector = { remove_variable = fs_legislation_preserved_ballot }
			}
			else_if = {
				limit = {
					scope:fs_legislation_elector = {
						NOT = { var:fs_legislation_ballot_seat = scope:fs_legislation_seat }
					}
				}
				# Discard votes on another held seat so changing the representative
				# cannot revive an older selection. Duplicate entries of this seat stay intact.
				scope:fs_legislation_seat = {
					set_variable = { name = fs_legislation_ballot value = 0 }
					remove_variable = fs_legislation_ballot_cast
				}
			}
		}
	}
	if = {
		limit = { OR = { var:fs_legislation_stage = flag:vote var:fs_legislation_stage = flag:revote } }
		every_in_list = {
			variable = fs_legislation_voters
			limit = { is_ai = yes NOT = { fs_legislation_has_ballot_trigger = yes } }
			fs_legislation_cast_ai_ballot_effect = yes
		}
	}
}

# Only this internal effect may update an AI's public position after persuasion.
# The refresh path calls it only for a fresh, uncast ballot; it never changes human votes.
fs_legislation_cast_ai_ballot_effect = {
	if = {
		limit = {
			is_ai = yes
			fs_legislation_member_trigger = yes
			fs_legislation_voting_capacity_trigger = yes
			title:k_koreanism = { OR = { var:fs_legislation_stage = flag:vote var:fs_legislation_stage = flag:revote } }
			any_held_title = { fs_legislation_valid_ballot_seat_trigger = yes }
		}
		if = {
			limit = { fs_legislation_support_score_value > 5 }
			var:fs_legislation_ballot_seat = { set_variable = { name = fs_legislation_ballot value = 1 } }
		}
		else_if = {
			limit = { fs_legislation_support_score_value < -5 }
			var:fs_legislation_ballot_seat = { set_variable = { name = fs_legislation_ballot value = -1 } }
		}
		else = { var:fs_legislation_ballot_seat = { set_variable = { name = fs_legislation_ballot value = 0 } } }
		var:fs_legislation_ballot_seat = { set_variable = { name = fs_legislation_ballot_cast value = yes } }
	}
}

# Capture immutable timer tokens. Absolute deadlines also protect against callbacks
# from an older phase or proposal, including callbacks restored from a save.
fs_legislation_schedule_phase_effect = {
	set_variable = { name = fs_legislation_deadline value = { value = current_date add = $DAYS$ } }
	save_scope_value_as = { name = fs_legislation_timer_serial value = var:fs_legislation_serial }
	save_scope_value_as = { name = fs_legislation_timer_phase value = var:fs_legislation_phase_number }
	trigger_event = { on_action = fs_legislation_timeout days = $DAYS$ }
}

# Clear a finished proposal only after its replacement has passed all public guards.
fs_legislation_clear_proposal_payload_effect = {
	remove_variable = fs_legislation_target
	remove_variable = fs_legislation_doctrine
	remove_variable = fs_legislation_law_subject
	remove_variable = fs_legislation_proposed_dual_office_policy
}

fs_legislation_begin_effect = {
	change_variable = { name = fs_legislation_serial add = 1 }
	set_variable = { name = fs_legislation_kind value = flag:$KIND$ }
	set_variable = { name = fs_legislation_proposer value = scope:fs_legislation_actor }
	set_variable = { name = fs_legislation_stage value = flag:debate }
	set_variable = { name = fs_legislation_phase_number value = 1 }
	remove_variable = fs_legislation_reconsidered
	remove_variable = fs_legislation_result
	remove_variable = fs_legislation_outcome_applied
	remove_variable = fs_legislation_enactment_authorized
	remove_variable = fs_legislation_failure_reason
	fs_legislation_refresh_electorate_effect = yes
	fs_legislation_schedule_phase_effect = { DAYS = fs_legislation_debate_days }
	fs_legislation_notify_effect = { MESSAGE = FS_LEGISLATION_NOTICE_PROPOSED }
}

fs_legislation_open_vote_effect = {
	set_variable = { name = fs_legislation_stage value = flag:$STAGE$ }
	set_variable = { name = fs_legislation_phase_number value = $PHASE$ }
	# A reconsideration is a new ballot even though the proposal serial stays the same.
	every_in_global_list = {
		variable = fs_koreanism_parliament_seats
		limit = { fs_koreanism_parliament_seat_trigger = yes }
		set_variable = { name = fs_legislation_ballot value = 0 }
		remove_variable = fs_legislation_ballot_cast
	}
	fs_legislation_schedule_phase_effect = { DAYS = fs_legislation_vote_days }
	fs_legislation_refresh_electorate_effect = yes
	if = {
		limit = { var:fs_legislation_stage = flag:revote }
		fs_legislation_notify_effect = { MESSAGE = FS_LEGISLATION_NOTICE_REVOTE }
	}
	else = { fs_legislation_notify_effect = { MESSAGE = FS_LEGISLATION_NOTICE_VOTE } }
}

fs_legislation_finish_effect = {
	remove_variable = fs_legislation_stage
	remove_variable = fs_legislation_phase_number
	remove_variable = fs_legislation_deadline
	remove_variable = fs_legislation_enactment_authorized
	set_variable = { name = fs_legislation_ai_proposal_cooldown value = yes days = fs_legislation_ai_proposal_days }
	# Keep the proposal payload, last counts and result for the history panel.
	fs_legislation_notify_effect = { MESSAGE = FS_LEGISLATION_NOTICE_RESULT }
}

fs_legislation_enact_effect = {
	set_variable = { name = fs_legislation_enactment_authorized value = yes }
	remove_variable = fs_legislation_outcome_applied
	fs_legislation_execute_outcome_effect = yes
	if = {
		limit = { NOT = { has_variable = fs_legislation_outcome_applied } }
		set_variable = { name = fs_legislation_result value = flag:lapsed }
	}
	fs_legislation_finish_effect = yes
}

fs_legislation_close_vote_effect = {
	fs_legislation_refresh_electorate_effect = yes
	set_variable = { name = fs_legislation_last_enrolled value = fs_legislation_enrolled_value }
	set_variable = { name = fs_legislation_last_yes value = fs_legislation_yes_value }
	set_variable = { name = fs_legislation_last_no value = fs_legislation_no_value }
	set_variable = { name = fs_legislation_last_abstain value = fs_legislation_abstain_value }
	set_variable = { name = fs_legislation_last_threshold value = fs_legislation_threshold_value }
	if = {
		limit = {
			fs_legislation_enrolled_value > 0
			fs_legislation_yes_value >= fs_legislation_threshold_value
		}
		if = {
			limit = { var:fs_legislation_kind = flag:law var:fs_legislation_stage = flag:vote }
			set_variable = { name = fs_legislation_stage value = flag:presidential }
			set_variable = { name = fs_legislation_phase_number value = 3 }
			fs_legislation_schedule_phase_effect = { DAYS = fs_legislation_presidential_days }
			fs_legislation_notify_effect = { MESSAGE = FS_LEGISLATION_NOTICE_PRESIDENTIAL }
		}
		else = { fs_legislation_enact_effect = yes }
	}
	else = {
		set_variable = { name = fs_legislation_result value = flag:rejected }
		fs_legislation_finish_effect = yes
	}
}

# Idempotent advancement uses a saved absolute game date, never a decrementing UI clock.
fs_legislation_advance_effect = {
	if = {
		limit = { fs_legislation_active_trigger = yes fs_legislation_days_left_value <= 0 }
		if = {
			limit = { var:fs_legislation_stage = flag:debate }
			fs_legislation_open_vote_effect = { STAGE = vote PHASE = 2 }
		}
		else_if = {
			limit = { OR = { var:fs_legislation_stage = flag:vote var:fs_legislation_stage = flag:revote } }
			fs_legislation_close_vote_effect = yes
		}
		else_if = {
			limit = { var:fs_legislation_stage = flag:presidential var:fs_legislation_kind = flag:law }
			fs_legislation_enact_effect = yes
		}
	}
}

fs_legislation_ai_daily_effect = {
	if = {
		limit = { fs_legislation_active_trigger = yes }
		if = {
			limit = {
				OR = { var:fs_legislation_stage = flag:debate var:fs_legislation_stage = flag:vote var:fs_legislation_stage = flag:revote }
			}
			random_in_list = {
				variable = fs_legislation_voters
				limit = {
					is_ai = yes
					fs_legislation_voting_capacity_trigger = yes
					NOT = { var:fs_legislation_lobby_cooldown = title:k_koreanism.var:fs_legislation_serial }
				}
				save_scope_as = fs_legislation_ai_actor
				title:k_koreanism = {
					random_in_list = {
						variable = fs_legislation_voters
						limit = { this != scope:fs_legislation_ai_actor fs_legislation_voting_capacity_trigger = yes }
						save_scope_as = target
						scope:fs_legislation_ai_actor = {
							if = { limit = { fs_legislation_support_score_value > 0 } fs_legislation_lobby_yes_effect = yes }
							else = { fs_legislation_lobby_no_effect = yes }
						}
					}
				}
			}
		}
		else_if = {
			limit = {
				var:fs_legislation_stage = flag:presidential
				fs_legislation_days_left_value <= 23
				holder ?= { is_ai = yes fs_legislation_promulgation_valid_trigger = yes }
			}
			holder = {
				if = { limit = { fs_legislation_support_score_value < 0 fs_legislation_veto_valid_trigger = yes } fs_legislation_veto_effect = yes }
				else = { fs_legislation_promulgate_effect = yes }
			}
		}
	}
	else_if = {
		limit = { NOT = { has_variable = fs_legislation_ai_proposal_cooldown } }
		set_variable = { name = fs_legislation_ai_proposal_cooldown value = yes days = fs_legislation_ai_proposal_days }
		random_list = {
			50 = {
				trigger = { exists = holder holder = { is_ai = yes fs_legislation_can_propose_trigger = yes } }
				holder = { fs_legislation_ai_proposal_effect = yes }
			}
			50 = {
				trigger = { any_in_list = { variable = fs_legislation_voters is_ai = yes fs_legislation_can_propose_trigger = yes } }
				random_in_list = {
					variable = fs_legislation_voters
					limit = { is_ai = yes fs_legislation_can_propose_trigger = yes }
					fs_legislation_ai_proposal_effect = yes
				}
			}
		}
	}
}

# Character scope. The body key is fixed before sending; only names and tallies
# are read from saved scopes when displaying an old notification.
fs_legislation_send_result_toast_effect = {
	send_interface_toast = {
		type = event_toast_text_neutral
		title = $MESSAGE$
		desc = fs_legislation_notice_$KIND$_$RESULT$
		left_icon = title:k_koreanism
	}
}

# Character scope. Each motion has one success outcome, rejection or lapse.
fs_legislation_send_result_notice_effect = {
	if = {
		limit = { title:k_koreanism = { var:fs_legislation_result = flag:$SUCCESS$ } }
		fs_legislation_send_result_toast_effect = { MESSAGE = $MESSAGE$ KIND = $KIND$ RESULT = $SUCCESS$ }
	}
	else_if = {
		limit = { title:k_koreanism = { var:fs_legislation_result = flag:rejected } }
		fs_legislation_send_result_toast_effect = { MESSAGE = $MESSAGE$ KIND = $KIND$ RESULT = rejected }
	}
	else = {
		fs_legislation_send_result_toast_effect = { MESSAGE = $MESSAGE$ KIND = $KIND$ RESULT = lapsed }
	}
}

fs_legislation_send_notice_effect = {
	if = {
		limit = {
			title:k_koreanism = {
				has_variable = fs_legislation_result
				NOT = { fs_legislation_active_trigger = yes }
			}
		}
		if = {
			limit = { title:k_koreanism = { var:fs_legislation_kind = flag:impeachment } }
			fs_legislation_send_result_notice_effect = { MESSAGE = $MESSAGE$ KIND = impeachment SUCCESS = impeached }
		}
		else_if = {
			limit = { title:k_koreanism = { var:fs_legislation_kind = flag:lift_martial_law } }
			fs_legislation_send_result_notice_effect = { MESSAGE = $MESSAGE$ KIND = lift_martial_law SUCCESS = martial_law_lifted }
		}
		else_if = {
			limit = { title:k_koreanism = { var:fs_legislation_kind = flag:censure } }
			fs_legislation_send_result_notice_effect = { MESSAGE = $MESSAGE$ KIND = censure SUCCESS = censured }
		}
		else_if = {
			limit = { title:k_koreanism = { var:fs_legislation_kind = flag:expulsion } }
			fs_legislation_send_result_notice_effect = { MESSAGE = $MESSAGE$ KIND = expulsion SUCCESS = expelled }
		}
		else_if = {
			limit = {
				title:k_koreanism = {
					fs_legislation_dual_office_law_trigger = yes
					var:fs_legislation_proposed_dual_office_policy = flag:allowed
				}
			}
			fs_legislation_send_result_notice_effect = { MESSAGE = $MESSAGE$ KIND = dual_office_allowed SUCCESS = enacted }
		}
		else_if = {
			limit = {
				title:k_koreanism = {
					fs_legislation_dual_office_law_trigger = yes
					var:fs_legislation_proposed_dual_office_policy = flag:forbidden
				}
			}
			fs_legislation_send_result_notice_effect = { MESSAGE = $MESSAGE$ KIND = dual_office_forbidden SUCCESS = enacted }
		}
		else = {
			fs_legislation_send_result_notice_effect = { MESSAGE = $MESSAGE$ KIND = law SUCCESS = enacted }
		}
	}
	else = {
		send_interface_toast = {
			title = $MESSAGE$
			left_icon = title:k_koreanism
			custom_tooltip = FS_LEGISLATION_NOTICE_INSTRUCTION
		}
	}
}

# Human members, the current president and the affected character receive one notice.
fs_legislation_notify_effect = {
	if = {
		limit = {
			has_variable = fs_legislation_result
			NOT = { fs_legislation_active_trigger = yes }
		}
		# Saved values keep an old notification independent of the next motion.
		save_scope_value_as = { name = fs_legislation_notice_enrolled value = var:fs_legislation_last_enrolled }
		save_scope_value_as = { name = fs_legislation_notice_threshold value = var:fs_legislation_last_threshold }
		save_scope_value_as = { name = fs_legislation_notice_yes value = var:fs_legislation_last_yes }
		save_scope_value_as = { name = fs_legislation_notice_no value = var:fs_legislation_last_no }
		save_scope_value_as = { name = fs_legislation_notice_abstain value = var:fs_legislation_last_abstain }
		var:fs_legislation_target ?= { save_scope_as = fs_legislation_notice_target }
	}
	every_in_list = {
		variable = fs_legislation_voters
		limit = { is_ai = no is_alive = yes }
		fs_legislation_send_notice_effect = { MESSAGE = $MESSAGE$ }
	}
	holder ?= {
		if = {
			limit = { is_ai = no is_alive = yes }
			save_scope_as = fs_legislation_notice_president
			if = {
				limit = {
					NOT = {
						title:k_koreanism = {
							is_target_in_variable_list = { name = fs_legislation_voters target = scope:fs_legislation_notice_president }
						}
					}
				}
				fs_legislation_send_notice_effect = { MESSAGE = $MESSAGE$ }
			}
		}
	}
	if = {
		limit = {
			has_variable = fs_legislation_result
			NOT = { fs_legislation_active_trigger = yes }
			exists = var:fs_legislation_target
		}
		var:fs_legislation_target = {
			if = {
				limit = {
					is_ai = no
					is_alive = yes
					NOT = { has_title = title:k_koreanism }
					NOT = {
						title:k_koreanism = {
							is_target_in_variable_list = { name = fs_legislation_voters target = scope:fs_legislation_notice_target }
						}
					}
				}
				# A deposed president or expelled MP still receives their own result.
				fs_legislation_send_notice_effect = { MESSAGE = $MESSAGE$ }
			}
		}
	}
}
