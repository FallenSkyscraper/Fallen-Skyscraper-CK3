# CB scope. Recheck immediately before transferring land so concurrent victories
# cannot remove the defender's final personally held county.
fs_seoul_restore_district_order_victory_effect = {
	if = {
		limit = { fs_seoul_district_order_war_valid_trigger = yes }
		every_in_list = {
			list = target_titles
			save_scope_as = fs_seoul_restored_county
		}
		create_character = {
			age = { 20 40 }
			location = scope:fs_seoul_restored_county.title_province
			culture = scope:fs_seoul_restored_county.culture
			faith = scope:fs_seoul_restored_county.faith
			rite = scope:fs_seoul_restored_county.rite
			dynasty = generate
			random_traits = yes
			gender_female_chance = {
				if = {
					limit = { scope:fs_seoul_restored_county.rite = { rite_has_doctrine = doctrine_gender_male_dominated } }
					add = 0
				}
				else_if = {
					limit = { scope:fs_seoul_restored_county.rite = { rite_has_doctrine = doctrine_gender_female_dominated } }
					add = 100
				}
				else = { add = 50 }
			}
			save_scope_as = fs_seoul_new_district_governor
		}
		create_title_and_vassal_change = {
			type = conquest
			save_scope_as = fs_seoul_district_title_change
			add_claim_on_loss = no
		}
		scope:fs_seoul_restored_county = {
			change_title_holder = {
				holder = scope:fs_seoul_new_district_governor
				change = scope:fs_seoul_district_title_change
				government_base = title:k_seoul.holder
				take_baronies = yes
			}
		}
		resolve_title_and_vassal_change = scope:fs_seoul_district_title_change
		# Resolve the grant before attaching the newly landed ruler to the mayor.
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = fs_seoul_district_liege_change
		}
		scope:fs_seoul_new_district_governor = {
			change_liege = {
				liege = title:k_seoul.holder
				change = scope:fs_seoul_district_liege_change
			}
		}
		resolve_title_and_vassal_change = scope:fs_seoul_district_liege_change
		scope:fs_seoul_new_district_governor = { change_government = seoul_government }
		scope:attacker = {
			change_influence = { value = fs_seoul_district_order_victory_influence }
			add_prestige = fs_seoul_district_order_victory_prestige
		}
		# The fixed 250 prestige above already rewards the primary attacker.
		modify_allies_of_participants_fame_values = {
			WINNER = scope:attacker
			FAME_BASE = minor_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}
	}
}

# Interaction scope. There is no send-time charge or global purchase cooldown.
fs_seoul_purchase_truce_accept_effect = {
	if = {
		limit = { fs_seoul_purchase_truce_valid_trigger = yes }
		scope:actor = {
			change_influence = { value = fs_seoul_purchase_truce_cost multiply = -1 }
			add_truce_both_ways = {
				character = scope:recipient
				days = fs_seoul_purchase_truce_days
				name = TRUCE_PURCHASED
			}
			add_to_bought_truce_list = { TARGET = scope:recipient DAYS = fs_seoul_purchase_truce_days }
			send_interface_toast = {
				type = event_toast_effect_good
				title = fs_seoul_purchase_truce_accepted_actor
				left_icon = scope:recipient
			}
		}
		scope:recipient = {
			change_influence = { value = fs_seoul_purchase_truce_recipient_influence }
			send_interface_toast = {
				type = event_toast_effect_good
				title = fs_seoul_purchase_truce_accepted_recipient
				left_icon = scope:actor
			}
		}
	}
	else = {
		scope:actor = {
			send_interface_toast = {
				type = event_toast_effect_neutral
				title = fs_seoul_purchase_truce_terms_changed
				left_icon = scope:recipient
			}
		}
	}
}
