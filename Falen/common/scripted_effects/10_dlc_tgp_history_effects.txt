tgp_make_military_celestial_provinces_effect = {
	holder ?= {
		if = {
			limit = { has_subject_contract_group = celestial_vassal }
			vassal_contract_set_obligation_level = {
				type = celestial_provinces
				level = 3 # Military
			}
			if = {
				limit = {
					NOT = {
						has_realm_law = celestial_military_appointment_succession_law
					}
				}
				add_realm_law_skip_effects = celestial_military_appointment_succession_law
			}
			every_vassal = {
				limit = {
					primary_title.tier >= tier_county
					has_subject_contract_group = celestial_vassal
				}
				vassal_contract_set_obligation_level = {
					type = celestial_provinces
					level = 3 # Military
				}
				if = {
					limit = {
						NOT = {
							has_realm_law = celestial_military_appointment_succession_law
						}
					}
					add_realm_law_skip_effects = celestial_military_appointment_succession_law
				}
				every_vassal = {
					limit = {
						primary_title.tier >= tier_county
						has_subject_contract_group = celestial_vassal
					}
					vassal_contract_set_obligation_level = {
						type = celestial_provinces
						level = 3 # Military
					}
					if = {
						limit = {
							NOT = {
								has_realm_law = celestial_military_appointment_succession_law
							}
						}
						add_realm_law_skip_effects = celestial_military_appointment_succession_law

			}
		}
	}
