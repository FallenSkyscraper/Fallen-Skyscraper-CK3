# Market visits transfer discoveries already known at their source.
# Source selection runs from the current situation and only considers its active markets.
fs_silk_road_receive_innovation_effect = {
	if = {
		limit = {
			scope:innovation_temp = {
				this = culture_innovation:innovation_$INNOVATION$
			}
			scope:innovation_temp = {
				is_known_by_culture = scope:market_temp.culture
			}
			NOT = {
				has_innovation = innovation_$INNOVATION$
			}
		}
		add_innovation = innovation_$INNOVATION$
	}
}
