# Character scope: the candidate. SEAT must be a vacant, pre-defined title.
fs_grant_koreanism_parliament_seat_effect = {
	if = {
		limit = {
			fs_koreanism_parliament_appointment_candidate_trigger = yes
			exists = title:k_koreanism.holder
			exists = $SEAT$
			title:k_koreanism.holder ?= { fs_koreanism_parliament_president_trigger = yes }
			$SEAT$ ?= { fs_koreanism_appointable_parliament_seat_trigger = yes }
		}
		fs_apply_koreanism_parliament_seat_effect = { SEAT = $SEAT$ ELECTED = no }
	}
}

# Character scope. Preserve the original primary heir before any government or title change.
# Claims are granted on directly held, claimable titles, never on a vassal's domain.
fs_protect_koreanism_parliament_entry_heir_effect = {
	if = {
		limit = {
			is_landed = yes
			exists = primary_heir
			primary_heir = { is_alive = yes this != prev }
		}
		primary_heir = { save_scope_as = fs_parliament_previous_primary_heir }
		every_held_title = {
			limit = { fs_koreanism_parliament_entry_claimable_title_trigger = yes }
			scope:fs_parliament_previous_primary_heir = { add_pressed_claim = prev }
		}
	}
}

# Internal transfer shared by authorized appointments and elections. ELECTED is yes/no.
# Callers validate authority/reservation before entering. Only elections may abdicate a lord;
# do not invoke directly from UI. Residence and activity are not rechecked after abdication.
fs_apply_koreanism_parliament_seat_effect = {
	if = {
		limit = {
			fs_koreanism_parliament_candidate_trigger = yes
			exists = $SEAT$
			$SEAT$ ?= { fs_koreanism_vacant_parliament_seat_trigger = yes }
			OR = {
				always = $ELECTED$
				fs_koreanism_parliament_appointment_candidate_trigger = yes
			}
		}
		save_scope_as = fs_parliament_recipient
		$SEAT$ = { save_scope_as = fs_parliament_seat }
		fs_protect_koreanism_parliament_entry_heir_effect = yes
		save_scope_value_as = { name = fs_parliament_entry_requires_abdication value = no }
		if = {
			limit = {
				always = $ELECTED$
				fs_koreanism_parliament_dual_office_forbidden_trigger = yes
				fs_koreanism_parliament_secular_lord_trigger = yes
			}
			save_scope_value_as = { name = fs_parliament_entry_requires_abdication value = yes }
			# Native deposition resolves each old title's succession, including partition.
			# The saved winner preserves identity, not player control: native deposition
			# may end a human ruler's game before the following parliamentary grant.
			depose = yes
		}
		scope:fs_parliament_recipient = {
			if = {
				limit = {
					fs_koreanism_parliament_candidate_trigger = yes
					scope:fs_parliament_seat = { fs_koreanism_vacant_parliament_seat_trigger = yes }
					OR = {
						scope:fs_parliament_entry_requires_abdication = no
						is_landed = no
					}
				}
				save_scope_as = fs_parliament_government_base
				# Elections do not depend on the president being available or consenting.
				if = {
					limit = { exists = title:k_koreanism.holder }
					title:k_koreanism.holder = { save_scope_as = fs_parliament_government_base }
				}
				scope:fs_parliament_seat = {
					if = {
						limit = {
							NOT = {
								faith:koreanism_faith = {
									any_clerical_elector_title = { this = scope:fs_parliament_seat }
								}
							}
						}
						add_clerical_elector = faith:koreanism_faith
					}
					create_title_and_vassal_change = {
						type = granted
						save_scope_as = fs_parliament_change
						add_claim_on_loss = no
					}
					change_title_holder = {
						holder = scope:fs_parliament_recipient
						change = scope:fs_parliament_change
						government_base = scope:fs_parliament_government_base
					}
					resolve_title_and_vassal_change = scope:fs_parliament_change
				}
				if = {
					limit = { scope:fs_parliament_seat = { holder = scope:fs_parliament_recipient } }
					add_character_flag = cardinal_flag
					fs_ensure_koreanism_government_effect = yes
				}
			}
		}
	}
}

# Character scope: the outgoing member. Static titles remain registered and reusable.
fs_vacate_koreanism_parliament_seat_effect = {
	if = {
		limit = {
			exists = $SEAT$
			$SEAT$ ?= {
				fs_koreanism_parliament_seat_trigger = yes
				holder = prev
			}
		}
		destroy_title = $SEAT$
		if = {
			limit = { NOT = { exists = clerical_elector_title } }
			remove_character_flag = cardinal_flag
		}
		fs_ensure_koreanism_government_effect = yes
	}
}


# Keep the native behavior for every title outside the fixed parliament.
pam_destroy_cardinal_title_effect = {
	if = {
		limit = { $CARDINALATE$ = { fs_koreanism_parliament_seat_trigger = yes } }
		fs_vacate_koreanism_parliament_seat_effect = { SEAT = $CARDINALATE$ }
	}
	else = {
		if = { # A dynamic cardinalate is deleted outright, so it has to leave the reuse pool with it
			limit = { $CARDINALATE$ = { exists = var:pam_dynamic_cardinalate } }
			remove_list_global_variable = {
				name = roma_cardinalates
				target = $CARDINALATE$
			}
		}
		remove_character_flag = cardinal_flag
		destroy_title = $CARDINALATE$
		if = {
			limit = {
				NOR = {
					exists = clerical_elector_title
					any_held_title = { has_clerical_region = yes }
				}
			}
			situation:the_christian_church ?= {
				if = {
					limit = { is_manual_participant = prev }
					remove_manual_participant = prev
				}
			}
		}
	}
}

# Koreanism has no minimum membership and never seeds or automatically replaces MPs.
pam_seed_cardinals_effect = {
	if = {
		limit = {
			NOT = { this = faith:koreanism_faith }
			has_pam_dlc_trigger = yes
			has_doctrine_parameter = has_clerical_electors
			any_clerical_elector_title = { count < 10 }
			OR = {
				exists = religious_head
				exists = title:k_papal_state.holder
			}
		}
		# Callers fire this from on_actions whose root is a character, so the faith needs its own scope
		save_scope_as = pam_seed_faith
		if = {
			limit = { exists = religious_head }
			religious_head = { save_scope_as = pam_seed_hof }
		}
		else = {
			title:k_papal_state.holder = { save_scope_as = pam_seed_hof }
		}
		scope:pam_seed_hof.capital_county ?= { save_scope_as = pam_seed_cardinal_county }
		if = {
			limit = { NOT = { exists = scope:pam_seed_cardinal_county } }
			title:c_roma = { save_scope_as = pam_seed_cardinal_county }
		}
		# One ordered pass - same eligibility as pam_cardinal_vacancy_on_death_effect
		ordered_clerical_region_title = {
			limit = {
				holder ?= {
					is_alive = yes
					is_adult = yes
					is_incapable = no
					NOT = { has_trait = excommunicated }
					NOT = { exists = clerical_elector_title }
					this != scope:pam_seed_hof
					pam_can_hold_cardinalate_trigger = yes
					pam_pope_is_head_of_faith_or_rite_trigger = yes
					pam_ai_cardinal_candidate_is_immersive_trigger = yes
				}
			}
			order_by = holder.piety_level
			max = 10
			check_range_bounds = no
			holder = {
				if = {
					limit = {
						scope:pam_seed_faith = { any_clerical_elector_title = { count < 10 } }
					}
					grant_cardinal_title_effect = {
						CARDINAL_COUNTY = capital_county
					}
					if = {
						limit = {
							is_ai = no
							NOT = { current_date_is_start_date_trigger = yes }
						}
						scope:pam_seed_hof = { save_scope_as = head_of_faith }
						pam_notify_player_became_cardinal_effect = yes
					}
					add_to_list = pam_seeded_cardinals
				}
			}
		}
		# Mint only what the promotion pass could not fill (count = 10 is a hard bound)
		scope:pam_seed_hof = {
			while = {
				count = 10
				limit = {
					scope:pam_seed_faith = { any_clerical_elector_title = { count < 10 } }
				}
				pam_create_seed_cardinal_effect = {
					HOF = scope:pam_seed_hof
					LOCATION = scope:pam_seed_cardinal_county.title_province
					FAITH = scope:pam_seed_faith
				}
				scope:pam_new_cardinal ?= {
					grant_cardinal_title_effect = {
						CARDINAL_COUNTY = scope:pam_seed_cardinal_county
					}
					add_to_list = pam_seeded_cardinals
				}
			}
		}
		# Seeding at game start is part of the setup, but mid-game it would otherwise be silent
		if = {
			limit = {
				NOT = { current_date_is_start_date_trigger = yes }
				any_in_list = { list = pam_seeded_cardinals count >= 1 }
				scope:pam_seed_hof = { is_ai = no }
			}
			ordered_in_list = {
				list = pam_seeded_cardinals
				order_by = piety_level
				max = 1
				check_range_bounds = no
				save_scope_as = cardinal_candidate
			}
			scope:pam_seed_hof = {
				trigger_event = {
					id = pam_clergy_events.0042
					days = 3
				}
			}
		}
	}
}

pam_cardinal_vacancy_on_death_effect = {
	if = {
		limit = {
			NOT = { faith = faith:koreanism_faith }
			has_pam_dlc_trigger = yes
			OR = {
				exists = clerical_elector_title
				has_character_flag = cardinal_flag
			}
			OR = {
				exists = faith.religious_head
				exists = title:k_papal_state.holder
			}
			faith = {
				has_doctrine_parameter = has_clerical_electors
				any_clerical_elector_title = { count < 10 }
			}
		}
		if = {
			limit = { exists = faith.religious_head }
			faith.religious_head = { save_scope_as = head_of_faith }
		}
		else = {
			title:k_papal_state.holder = { save_scope_as = head_of_faith }
		}
		if = {
			limit = {
				scope:head_of_faith = { is_ai = yes }
			}
			faith = {
				ordered_clerical_region_title = {
					limit = {
						holder ?= {
							is_alive = yes
							is_adult = yes
							is_incapable = no
							NOT = { has_trait = excommunicated }
							NOT = { exists = clerical_elector_title }
							this != scope:head_of_faith
							pam_can_hold_cardinalate_trigger = yes
							pam_pope_is_head_of_faith_or_rite_trigger = yes
							pam_ai_cardinal_candidate_is_immersive_trigger = yes
							pam_is_antipope_trigger = no
						}
					}
					order_by = holder.piety_level
					max = 1
					check_range_bounds = no
					holder = {
						save_scope_as = cardinal_candidate
						trigger_event = {
							id = pam_clergy_events.0040
							days = 3
						}
					}
				}
			}
		}
		else = {
			faith = {
				ordered_clerical_region_title = {
					limit = {
						holder ?= {
							is_ai = yes
							is_alive = yes
							is_adult = yes
							is_incapable = no
							NOT = { has_trait = excommunicated }
							NOT = { exists = clerical_elector_title }
							this != scope:head_of_faith
							pam_can_hold_cardinalate_trigger = yes
							pam_pope_is_head_of_faith_or_rite_trigger = yes
							pam_ai_cardinal_candidate_is_immersive_trigger = yes
							pam_is_antipope_trigger = no
						}
					}
					order_by = holder.piety_level
					max = 1
					check_range_bounds = no
					holder = {
						save_scope_as = cardinal_candidate
					}
				}
			}
			if = {
				limit = {
					exists = scope:cardinal_candidate
				}
				scope:head_of_faith = {
					trigger_event = {
						id = pam_clergy_events.0041
						days = 3
					}
				}
			}
		}
	}
}
