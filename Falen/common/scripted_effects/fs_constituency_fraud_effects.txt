# Activity scope, creation only. Older already-running activities stay on old rules.
fs_constituency_enable_fraud_rules_effect = {
	if = {
		limit = { NOT = { has_variable = fs_constituency_fraud_rules } }
		set_variable = { name = fs_constituency_fraud_rules value = yes }
		set_variable = { name = fs_constituency_public_fraud_count value = 0 }
		set_variable = { name = fs_constituency_covered_fraud_count value = 0 }
		scope:host.rite ?= { save_scope_as = fs_constituency_organizing_rite }
		if = {
			limit = { exists = scope:fs_constituency_organizing_rite }
			set_variable = { name = fs_constituency_organizing_rite value = scope:fs_constituency_organizing_rite }
		}
	}
}

# Character scope. Arrival, passive pulses and roster freeze share this offer guard.
fs_constituency_offer_integrity_choice_effect = {
	if = {
		limit = {
			scope:activity = {
				has_variable = fs_constituency_fraud_rules
				fs_constituency_running_activity_trigger = yes
				NOT = { has_variable = fs_constituency_integrity_locked }
			}
			has_activity_intent = fs_constituency_candidate_intent
			fs_constituency_candidate_for_seat_trigger = { SEAT = scope:activity.var:fs_constituency_seat }
			involved_activity = scope:activity
			location = scope:activity.activity_location
		}
		save_scope_as = fs_constituency_integrity_candidate
		if = {
			limit = {
				scope:activity = {
					NOT = { is_target_in_variable_list = { name = fs_constituency_integrity_offered target = scope:fs_constituency_integrity_candidate } }
				}
			}
			scope:activity = {
				save_scope_as = fs_constituency_fraud_activity
				add_to_variable_list = { name = fs_constituency_integrity_offered target = scope:fs_constituency_integrity_candidate }
			}
			trigger_event = fs_constituency_fraud.1
		}
	}
}

# Activity scope. All timers capture a round; no timer can change a later round.
fs_constituency_start_fraud_round_effect = {
	if = {
		limit = { fs_constituency_fraud_round_open_trigger = yes }
		clear_variable_list = fs_constituency_evidence
		clear_variable_list = fs_constituency_case_observers
		clear_variable_list = fs_constituency_case_suspects
		clear_variable_list = fs_constituency_observer_done
		clear_variable_list = fs_constituency_host_pending
		clear_variable_list = fs_constituency_case_resolved
		save_scope_as = fs_constituency_fraud_activity
		save_scope_value_as = { name = fs_constituency_fraud_round value = var:fs_constituency_round }
		if = {
			limit = { var:fs_constituency_round = 1 }
			scope:host = { trigger_event = { id = fs_constituency_fraud.2 days = fs_constituency_integrity_grace_days } }
		}
		else = { fs_constituency_award_fraud_bonuses_effect = yes }
		scope:host = { trigger_event = { id = fs_constituency_fraud.3 days = fs_constituency_fraud_audit_days } }
	}
}

# Activity scope. Called by the first-day timer, and defensively by round close.
fs_constituency_lock_integrity_effect = {
	if = {
		limit = { fs_constituency_fraud_round_open_trigger = yes }
		if = {
			limit = { NOT = { has_variable = fs_constituency_integrity_locked } }
			set_variable = { name = fs_constituency_integrity_locked value = yes }
			every_in_list = {
				variable = fs_constituency_candidates
				limit = {
					NOT = { scope:activity = { is_target_in_variable_list = { name = fs_constituency_fraud_candidates target = prev } } }
					NOT = { scope:activity = { is_target_in_variable_list = { name = fs_constituency_honest_candidates target = prev } } }
				}
				save_scope_as = fs_constituency_integrity_candidate
				scope:activity = { add_to_variable_list = { name = fs_constituency_honest_candidates target = scope:fs_constituency_integrity_candidate } }
			}
		}
		fs_constituency_award_fraud_bonuses_effect = yes
	}
}

# Character scope, numeric method snapshot. No public log exposes hidden cheating.
fs_constituency_select_fraud_method_effect = {
	if = {
		limit = { scope:activity = { var:fs_constituency_round = 1 } }
		random_list = {
			0 = { modifier = { add = fs_constituency_fraud_variant_weight } save_scope_value_as = { name = fs_constituency_case_method value = 1 } }
			0 = { modifier = { add = fs_constituency_fraud_variant_weight } save_scope_value_as = { name = fs_constituency_case_method value = 2 } }
		}
	}
	else_if = {
		limit = { scope:activity = { var:fs_constituency_round = 2 } }
		save_scope_value_as = { name = fs_constituency_case_method value = 3 }
	}
	else = {
		random_list = {
			0 = { modifier = { add = fs_constituency_fraud_variant_weight } save_scope_value_as = { name = fs_constituency_case_method value = 4 } }
			0 = { modifier = { add = fs_constituency_fraud_variant_weight } save_scope_value_as = { name = fs_constituency_case_method value = 5 } }
		}
	}
}

fs_constituency_award_fraud_bonuses_effect = {
	if = {
		limit = {
			fs_constituency_fraud_round_open_trigger = yes
			OR = { NOT = { has_variable = fs_constituency_bonus_round } var:fs_constituency_bonus_round < var:fs_constituency_round }
		}
		set_variable = { name = fs_constituency_bonus_round value = var:fs_constituency_round }
		save_scope_as = fs_constituency_fraud_activity
		save_scope_value_as = { name = fs_constituency_fraud_round value = var:fs_constituency_round }
		every_in_list = {
			variable = fs_constituency_candidates
			limit = { fs_constituency_registered_candidate_trigger = { ACTIVITY = scope:activity } }
			if = {
				limit = {
					scope:activity = { var:fs_constituency_administration = flag:fs_constituency_biased }
					rite = scope:activity.var:fs_constituency_organizing_rite
				}
				change_variable = { name = fs_constituency_score add = fs_constituency_bias_gain }
			}
			if = {
				limit = { scope:activity = { is_target_in_variable_list = { name = fs_constituency_fraud_candidates target = prev } } }
				change_variable = { name = fs_constituency_score add = fs_constituency_fraud_gain }
				fs_constituency_select_fraud_method_effect = yes
				set_variable = { name = fs_constituency_fraud_method value = scope:fs_constituency_case_method }
				trigger_event = fs_constituency_fraud.10
			}
		}
	}
}

# Activity scope. First find evidence independently for each unexposed cheater.
fs_constituency_audit_fraud_effect = {
	if = {
		limit = {
			fs_constituency_fraud_round_open_trigger = yes
			OR = { NOT = { has_variable = fs_constituency_audited_round } var:fs_constituency_audited_round < var:fs_constituency_round }
		}
		set_variable = { name = fs_constituency_audited_round value = var:fs_constituency_round }
		every_in_list = {
			variable = fs_constituency_fraud_candidates
			limit = {
				fs_constituency_registered_candidate_trigger = { ACTIVITY = scope:activity }
				NOT = { scope:activity = { is_target_in_variable_list = { name = fs_constituency_public_offenders target = prev } } }
			}
			save_scope_as = fs_constituency_evidence_candidate
			random_list = {
				0 = {
					modifier = { add = fs_constituency_evidence_chance_value }
					scope:activity = { add_to_variable_list = { name = fs_constituency_evidence target = scope:fs_constituency_evidence_candidate } }
				}
				0 = { modifier = { add = fs_constituency_evidence_failure_value } }
			}
		}
		# Allocate real cases first. A limited monitor pool is meaningful oversight.
		every_attending_character = {
			limit = { fs_constituency_fraud_observer_trigger = yes }
			fs_constituency_assign_fraud_case_effect = yes
		}
	}
}

# Character scope. Clear previous loop scopes before attempting an empty random pick.
fs_constituency_assign_fraud_case_effect = {
	clear_saved_scope = fs_constituency_case_suspect
	save_scope_value_as = { name = fs_constituency_case_real value = 0 }
	save_scope_value_as = { name = fs_constituency_case_method value = 0 }
	save_scope_as = fs_constituency_case_observer
	scope:activity = {
		save_scope_as = fs_constituency_case_activity
		save_scope_value_as = { name = fs_constituency_case_round value = var:fs_constituency_round }
		random_in_list = {
			variable = fs_constituency_evidence
			limit = {
				fs_constituency_registered_candidate_trigger = { ACTIVITY = scope:activity }
				NOT = { scope:activity = { is_target_in_variable_list = { name = fs_constituency_case_suspects target = prev } } }
			}
			save_scope_as = fs_constituency_case_suspect
			save_scope_value_as = { name = fs_constituency_case_real value = 1 }
			save_scope_value_as = { name = fs_constituency_case_method value = var:fs_constituency_fraud_method }
		}
	}
	if = {
		limit = { NOT = { exists = scope:fs_constituency_case_suspect } }
		random_list = {
			0 = {
				modifier = { add = fs_constituency_false_suspicion_chance }
				scope:activity = {
					random_in_list = {
						variable = fs_constituency_candidates
						limit = {
							fs_constituency_registered_candidate_trigger = { ACTIVITY = scope:activity }
							NOT = { scope:activity = { is_target_in_variable_list = { name = fs_constituency_fraud_candidates target = prev } } }
							NOT = { scope:activity = { is_target_in_variable_list = { name = fs_constituency_case_suspects target = prev } } }
						}
						save_scope_as = fs_constituency_case_suspect
						fs_constituency_select_fraud_method_effect = yes
					}
				}
			}
			0 = { modifier = { add = fs_constituency_false_suspicion_failure_value } }
		}
	}
	if = {
		limit = { exists = scope:fs_constituency_case_suspect }
		scope:activity = {
			add_to_variable_list = { name = fs_constituency_case_suspects target = scope:fs_constituency_case_suspect }
			add_to_variable_list = { name = fs_constituency_case_observers target = scope:fs_constituency_case_observer }
		}
		trigger_event = fs_constituency_fraud.20
	}
}

# Activity scope. Resolve before invoking title/attendance callbacks. Conviction
# remains enforceable if a proved offender leaves while a host reply is pending.
fs_constituency_punish_suspect_effect = {
	if = {
		limit = {
			is_target_in_variable_list = { name = fs_constituency_public_offenders target = scope:fs_constituency_case_suspect }
			NOT = { is_target_in_variable_list = { name = fs_constituency_case_resolved target = scope:fs_constituency_case_suspect } }
		}
		add_to_variable_list = { name = fs_constituency_case_resolved target = scope:fs_constituency_case_suspect }
		fs_constituency_ensure_fraud_law_effect = yes
		scope:fs_constituency_case_suspect = {
			if = {
				limit = { is_alive = yes }
				add_trait = fs_election_offender
				add_prestige_level = fs_constituency_offender_level_loss
				add_piety_level = fs_constituency_offender_level_loss
				if = {
					limit = { var:fs_constituency_candidate_activity = scope:activity }
					set_variable = { name = fs_constituency_disqualified value = yes }
					set_variable = { name = fs_constituency_fraud_disqualified value = yes }
				}
				if = {
					limit = { involved_activity = scope:activity this != scope:activity.activity_host }
					remove_from_activity = scope:activity
				}
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = fs_constituency_fraud_conviction_title
					left_icon = scope:fs_constituency_case_suspect
					custom_tooltip = fs_constituency_fraud_conviction_tt
				}
			}
		}
		add_activity_log_entry = {
			key = fs_constituency_fraud_punished
			character = scope:fs_constituency_case_suspect
			location = scope:activity.activity_location
			score = 100
			tags = { activity }
			show_in_conclusion = yes
		}
	}
}
