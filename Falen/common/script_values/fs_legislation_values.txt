	value = fs_legislation_enrolled_value

		limit = {
			title:k_koreanism = {

# Character scope. Unpersuaded assessment: laws prioritize policy over personal loyalty.
# Law opinion contributes at most +/-10; social ties are deliberately below policy deltas.
# Motions keep the existing personal/accountability model because their target is a person.
# AI lobbying uses this value so accumulated persuasion cannot reverse its campaign direction.
fs_legislation_base_support_score_value = {
	value = 0
	if = {
		limit = { title:k_koreanism = { var:fs_legislation_kind = flag:law } }
		add = { value = fs_legislation_ai_pending_law_change_value multiply = fs_legislation_ai_law_preference_multiplier }
		if = {
			limit = { title:k_koreanism = { fs_legislation_tenet_law_trigger = yes } }
			if = { limit = { title:k_koreanism = { var:fs_legislation_tenet_status = flag:permitted } } add = fs_legislation_ai_tenet_support_value }
			else_if = { limit = { title:k_koreanism = { var:fs_legislation_tenet_status = flag:prohibited } } subtract = fs_legislation_ai_tenet_support_value }
		}
		add = fs_legislation_ai_law_amendment_bias_value
		if = {
			limit = { exists = title:k_koreanism.var:fs_legislation_proposer }
			add = { value = "opinion(title:k_koreanism.var:fs_legislation_proposer)" divide = 10 min = -10 max = 10 }
			if = { limit = { this = title:k_koreanism.var:fs_legislation_proposer } add = 10 }
