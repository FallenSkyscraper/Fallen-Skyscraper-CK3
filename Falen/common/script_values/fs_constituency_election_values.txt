# Native guest-subset sorting uses a script value, as in imperial_examination.
fs_constituency_scoreboard_sort_value = {
	value = 0
	if = {
		limit = { has_variable = fs_constituency_score }
		add = var:fs_constituency_score
	}
	if = {
		limit = { OR = { has_variable = fs_constituency_disqualified is_alive = no } }
		subtract = 10000
	}
}
