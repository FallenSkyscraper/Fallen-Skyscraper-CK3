## Republic types AI values
ai_will_do_republic_type_communal_value = {
	value = 0

	# Extra scoring to make sure we are more likely to keep the current type, and not swap back and forth
	if = {
		limit = { has_realm_law = republic_type_communal }
		add = gdp_ai_keep_republic_type_extra_score_value
	}
}

	value = 0

}

ai_will_do_republic_type_maritime_value = {
	value = 0
	if = {
		limit = {
			highest_held_title_tier >= tier_kingdom
		}
		add = 1
	}
	if = {
		limit = {
			capital_province ?= {
				is_coastal = yes
			}
		}
		add = 2
	}
	else_if = {
		limit = {
			capital_county ?= {
				OR = {
					is_coastal_county = yes
					AND = {
						is_riverside_county = yes
						any_neighboring_county = {
							is_coastal_county = yes
						}
					}
				}
			}
		}
		add = 1
	}

	# Cultural reasons
	if = {
		limit = {
			culture ?= {
				has_cultural_tradition = tradition_maritime_mercantilism
			}
		}
		add = 1
	}

	# Historical maritime republics get extra score
	if = {
		limit = {
			gdp_is_historical_maritime_republic = yes
		}
		add = gdp_ai_historical_republic_extra_score_value
	}

	# Extra scoring to make sure we are more likely to keep the current type, and not swap back and forth
	if = {
		limit = { has_realm_law = republic_type_maritime }
		add = gdp_ai_keep_republic_type_extra_score_value
	}
}

## Men-at-arms AI values
gdp_ai_landless_republic_min_net_gold_income_value = 1.5

# Share of income a landless republic may spend on men-at-arms and still keep the minimum net gold
gdp_ai_landless_republic_maa_expense_gold_share_value = {
	value = 0
	if = {
		limit = { monthly_character_income > 0 }
		value = monthly_character_income
		subtract = monthly_character_expenses
		add = monthly_character_men_at_arms_expense_gold
		subtract = gdp_ai_landless_republic_min_net_gold_income_value
		divide = monthly_character_income
	}
	min = 0
}

gdp_ai_landless_republic_maa_ideal_share_value = 0.25

gdp_ai_landless_republic_maa_expense_gold_ideal_share_value = {
	value = gdp_ai_landless_republic_maa_expense_gold_share_value
	max = gdp_ai_landless_republic_maa_ideal_share_value
}

## Term Length AI values
law_ai_will_do_republic_assembly_term_length_1_value = {
	value = 0
	# The AI will only seek out the shortest term length if they greatly dislike the leading candidate
	if = {
		limit = {
			highest_liege_matching_rule:uses_assembly = {
				OR = { # For this one, we allow the AI to skip a level and go straight here when applicable
					has_realm_law = republic_assembly_term_length_2
					has_realm_law = republic_assembly_term_length_3
				}
			}
			house ?= {
				NOT = {
					has_house_aspiration_parameter = republic_signorial
				}
			}
			exists = highest_liege_matching_rule:uses_assembly.primary_heir
			OR = {
				has_relation_rival = highest_liege_matching_rule:uses_assembly.primary_heir
				opinion = {
					target = highest_liege_matching_rule:uses_assembly.primary_heir
					value <= gdp_disliked_republic_candidate_threshold_value
				}
			}
		}
		add = 4

		# Increase the chance if we have the Agitation ambition
		if = {
			limit = {
				house ?= { has_house_aspiration_parameter = republic_agitation }
			}
			add = 4
		}
	}
	
	# Grace period
	multiply = ai_law_grace_period_value
}

			add = 1
		}

			limit = {
				highest_liege_matching_rule:uses_assembly = {
					AND = {
						OR = {
							has_realm_law = republic_banishment_2_permitted_by_decree
							has_realm_law = republic_banishment_3_unrestricted
						}
						OR = {
							has_realm_law = republic_execution_2_permitted_by_decree
							has_realm_law = republic_execution_3_unrestricted
						}
					}
				}
			}
			add = 1
		}
		if = {
			limit = {
				highest_liege_matching_rule:uses_assembly = {
					AND = {
						has_realm_law = republic_banishment_3_unrestricted
						has_realm_law = republic_execution_3_unrestricted
					}
				}
			}
			add = 1
		}
		if = {
			limit = {
				highest_liege_matching_rule:uses_assembly = {
					AND = {
						has_realm_law = republic_title_revocation_2_allowed
						has_realm_law = republic_retract_vassal_2_allowed
					}
				}
			}
			add = 1
		}
		if = {
			limit = {
				highest_liege_matching_rule:uses_assembly = {
					has_realm_law = republic_vassal_refusal_2_treason
				}
			}
			add = 1
		}
	}
	
	# The AI wants to decrease it
	if = {
		limit = {
			highest_liege_matching_rule:uses_assembly = {
				has_realm_law = republic_assembly_term_length_3
			}
			house ?= {
				NOR = {
					has_house_aspiration_parameter = republic_signorial
					has_house_aspiration_parameter = republic_aggrandizement
				}
			}
		}

		if = {
			limit = {
				house ?= {
					OR = {
						has_house_aspiration_parameter = republic_communal
						has_house_aspiration_parameter = republic_commerce
					}
				}
			}
			add = 1
		}

		# The republic have "communal" favored laws
		if = {
			limit = {
				highest_liege_matching_rule:uses_assembly = {
					OR = {
						has_realm_law = republic_assembly_composition_general
						has_realm_law = republic_assembly_composition_parliament
						has_realm_law = republic_assembly_composition_roman_senate
					}
				}
			}
			add = 1
		}
		if = {
			limit = {
				highest_liege_matching_rule:uses_assembly = {
					has_realm_law = republic_assembly_authority_constitutional
				}
			}
			add = 1
		}

		# We greatly dislike the current candidate
		if = {
			limit = {
				highest_liege_matching_rule:uses_assembly = {
					has_realm_law = republic_assembly_term_length_3
				}
				exists = highest_liege_matching_rule:uses_assembly.primary_heir

					opinion = {
						target = highest_liege_matching_rule:uses_assembly.primary_heir
						value <= gdp_disliked_republic_candidate_threshold_value
					}
				}
			}
			add = 5
		}
	}

	# Grace period
	multiply = ai_law_grace_period_value
